def abbreviate_text(text):
    """Replace substrings in text using CONFIG_SHORTCUTS (longer keys first)."""
    if not isinstance(text, str):
        return text
    for long_str in sorted(CONFIG_SHORTCUTS, key=len, reverse=True):
        abbr = CONFIG_SHORTCUTS[long_str]
        if long_str in text:
            text = text.replace(long_str, abbr)
    return text

def apply_field_shortcuts(obj):
    """
    Recursively replace dictionary keys and string values using shortcuts.
    E.g. if 'archetype_details' -> 'ad', or '_type' -> 'T', etc.
    """
    if isinstance(obj, dict):
        new_dict = {}
        for k, v in obj.items():
            new_key = CONFIG_SHORTCUTS.get(k, k)
            new_dict[new_key] = apply_field_shortcuts(v)
        return new_dict
    elif isinstance(obj, list):
        return [apply_field_shortcuts(item) for item in obj]
    elif isinstance(obj, str):
        return abbreviate_text(obj)
    else:
        return obj

from dateutil.parser import parse as parse_date
import logging

def convert_field_value(field):
    """
    Convert recognized DV_* structures' 'value' (or 'magnitude') to a Python-friendly type
    (datetime, int, float) but keep the entire structure. For example:

      {
        "_type": "DV_DATE_TIME",
        "value": "2024-11-06T18:06:13.310089"
      }

    becomes:

      {
        "_type": "DV_DATE_TIME",
        "value": datetime.datetime(2024, 11, 6, 18, 6, 13, 310089)
      }

    so in Mongo you get an actual BSON Date. Meanwhile, you preserve _type and any
    subfields. No data is lost; only the 'value' is replaced by its parsed form.
    """
    if isinstance(field, dict):
        clone = dict(field)  # Shallow copy so we can modify safely

        # Identify if it matches either the "short" shape (T/v) or original shape (_type/value).
        if "T" in clone and "v" in clone:
            # short shape, e.g. {"T": "DV_DATE_TIME", "v": "2023-01-01T00:00:00Z"}
            type_code = clone["T"]
            val_key = "v"
        elif "_type" in clone and "value" in clone:
            # normal shape, e.g. {"_type": "DV_DATE_TIME", "value": "..."}
            type_code = clone["_type"]
            val_key = "value"
        else:
            # Not recognized as a DV_* object, so just recurse deeper
            for k, v in clone.items():
                clone[k] = convert_field_value(v)
            return clone

        try:
            # If it's a date/time type, parse only the 'value' field
            if type_code in ["ddt","DV_DATE_TIME","dd","DV_DATE","dti","DV_TIME"]:
                if val_key in clone:
                    raw_str = clone[val_key]
                    dt = parse_date(raw_str)  # dateutil parser => Python datetime
                    clone[val_key] = dt       # store the datetime object
                return clone

            # If it's a quantity or count, convert the numeric fields but keep all subfields
            elif type_code in ["dq","dc","DV_QUANTITY","DV_COUNT"]:
                # DV_QUANTITY often has 'magnitude' and 'units'
                if "magnitude" in clone:
                    mag_str = clone["magnitude"]
                    # parse float or int
                    if isinstance(mag_str, str) and '.' in mag_str:
                        clone["magnitude"] = float(mag_str)
                    else:
                        clone["magnitude"] = int(mag_str)
                # Some code uses "value" for the quantity
                elif val_key in clone:
                    val_str = clone[val_key]
                    if isinstance(val_str, str) and '.' in val_str:
                        clone[val_key] = float(val_str)
                    else:
                        clone[val_key] = int(val_str)
                return clone

            # If it's DV_ORDINAL
            elif type_code in ["DV_ORDINAL","DO"]:
                if val_key in clone:
                    clone[val_key] = int(clone[val_key])
                return clone

            # For other DV_* types (like DV_CODED_TEXT), we do not forcibly
            # reduce them, so we just return them as-is:
            return clone

        except Exception as e:
            logging.debug(f"Parsing '{field}' failed: {e}")
            # If anything goes wrong, we keep the structure as-is
            return clone

    elif isinstance(field, list):
        # recurse into each item
        return [convert_field_value(item) for item in field]

    else:
        # simple scalar => no special transform
        return field

def get_archetype_id(node_dict):
    """Extract archetype id from node (checking canonical keys)."""
    if not isinstance(node_dict, dict):
        return None
    if "archetype_details" in node_dict and isinstance(node_dict["archetype_details"], dict):
        maybe_id = node_dict["archetype_details"].get("archetype_id", {})
        return maybe_id.get("value")
    return node_dict.get("archetype_node_id")

ALLOWED_RM_TYPES = {
    "COMPOSITION", "SECTION", "ADMIN_ENTRY",
    "OBSERVATION", "EVALUATION", "INSTRUCTION",
    "ACTION", "CLUSTER", "ITEM_TREE", "ITEM_LIST",
    "ITEM_SINGLE", "ITEM_TABLE", "ELEMENT", "HISTORY",
    "EVENT", "POINT_EVENT", "INTERVAL_EVENT", "ACTIVITY",
    "ISM_TRANSITION", "INSTRUCTION_DETAILS", "CARE_ENTRY",
    "PARTY_PROXY", "EVENT_CONTEXT"
}

def is_openehr_node(node_dict):
    if not isinstance(node_dict, dict):
        return False
    rm_type = node_dict.get("_type", "")
    if rm_type in ALLOWED_RM_TYPES:
        return True
    if "archetype_details" in node_dict or "archetype_node_id" in node_dict:
        return True
    return False

def contains_openehr_node(obj):
    """
    Recursively checks if obj or any of its children is an openEHR node.
    """
    if is_openehr_node(obj):
        return True
    if isinstance(obj, dict):
        for v in obj.values():
            if contains_openehr_node(v):
                return True
        return False
    if isinstance(obj, list):
        for item in obj:
            if contains_openehr_node(item):
                return True
        return False
    return False

# ----------------------------
# Extract Archetype ID from Canonical JSON
# ----------------------------
def extract_archetype_id_from_canonical(canonical):
    """
    Recursively traverse canonical JSON to obtain the composition archetype id (full form).
    """
    if not canonical:
        return None
    if isinstance(canonical, dict):
        if "archetype_details" in canonical:
            td = canonical.get("archetype_details")
            if isinstance(td, dict):
                aid = td.get("archetype_id")
                if isinstance(aid, dict):
                    return aid.get("value")
                return aid
        for v in canonical.values():
            aid = extract_archetype_id_from_canonical(v)
            if aid:
                return aid
    elif isinstance(canonical, list):
        for item in canonical:
            aid = extract_archetype_id_from_canonical(item)
            if aid:
                return aid
    return None

# ----------------------------
# Flattening the Composition
# ----------------------------
def flatten_node(node, path="canonicalJSON", archetype_path="canonicalJSON", ancestors=None):
    """
    Recursively walk the composition tree. For each recognized openEHR node:
      - gather immediate data (non-nested openEHR structures) into node_data
      - parse them to native types
      - append to the flattened list
      - then recurse into subnodes
    """
    if ancestors is None:
        ancestors = []
    flattened = []
    recognized = is_openehr_node(node)

    if recognized:
        # gather "terminal" fields that do not contain sub openEHR nodes
        node_data = {}
        if isinstance(node, dict):
            for key, val in node.items():
                if contains_openehr_node(val):
                    # skip nested openEHR data here; that'll be handled by recursion
                    continue
                node_data[key] = val

        # Convert DV_... fields into Python datetime/numbers, etc.
        node_data = convert_field_value(node_data)

        flattened_entry = {
            "p": archetype_path,  # The 'path'
            "d": node_data
        }
        # Only add "a" if ancestors is non-empty
        if ancestors:
            flattened_entry["a"] = ancestors

        flattened.append(flattened_entry)

        # add this node's archetype ID to the ancestor chain
        this_id = get_archetype_id(node)
        if this_id:
            child_ancestors = ancestors + [this_id]
        else:
            child_ancestors = ancestors
    else:
        # if not recognized, ancestor chain stays the same
        child_ancestors = ancestors

    # Now recurse deeper
    if isinstance(node, dict):
        for key, val in node.items():
            new_path = f"{path}.{key}"
            if isinstance(val, dict):
                if is_openehr_node(val) or contains_openehr_node(val):
                    child_archetype_id = get_archetype_id(val)
                    if child_archetype_id:
                        child_archetype_path = f"{archetype_path}.{key}[{child_archetype_id}]"
                    else:
                        child_archetype_path = f"{archetype_path}.{key}"
                    flattened += flatten_node(val, new_path, child_archetype_path, child_ancestors)
            elif isinstance(val, list):
                for idx, item in enumerate(val):
                    item_path = f"{new_path}[{idx}]"
                    if is_openehr_node(item) or contains_openehr_node(item):
                        child_archetype_id = get_archetype_id(item)
                        if child_archetype_id:
                            item_archetype_path = f"{archetype_path}.{key}[{child_archetype_id}]"
                        else:
                            item_archetype_path = f"{archetype_path}.{key}[{idx}]"
                        flattened += flatten_node(item, item_path, item_archetype_path, child_ancestors)

    elif isinstance(node, list):
        for idx, item in enumerate(node):
            item_path = f"{path}[{idx}]"
            if is_openehr_node(item) or contains_openehr_node(item):
                child_archetype_id = get_archetype_id(item)
                if child_archetype_id:
                    item_archetype_path = f"{archetype_path}[{child_archetype_id}]"
                else:
                    item_archetype_path = f"{archetype_path}[{idx}]"
                flattened += flatten_node(item, item_path, item_archetype_path, child_ancestors)

    return flattened

def process_document(doc):
    """
    Process a single source document:
      1. Flatten canonicalJSON into nodes (cn).
      2. Extract the composition archetype id from canonical JSON.
      3. Return a dict with:
         {
            "ehr_id": doc.get("ehr_id"),
            "template_id": <archetype_id>,
            "nodes": [ ...flattened... ],
            "original": doc
         }
    """
    canonical = doc.get("canonicalJSON", {})
    archetype_id = extract_archetype_id_from_canonical(canonical)
    nodes = flatten_node(canonical, "canonicalJSON", "canonicalJSON", [])
    return {
        "ehr_id": doc.get("ehr_id", None),
        "template_id": archetype_id,
        "nodes": nodes,
        "original": doc
    }

def get_value_by_path(obj, path_segments):
    """
    Recursively retrieve all values in obj that match the nested path described by path_segments.
    - If we hit a dict, we look up the next segment as a key.
    - If we hit a list, we recurse into each item of the list.
    - If a segment is missing, we return None.
    - Potentially returns a list of values if the path matches multiple elements in a list.
    """
    if not path_segments:
        return obj  # No more segments to traverse, return current object

    key = path_segments[0]
    rest = path_segments[1:]

    if isinstance(obj, dict):
        # If current object is a dict, we just descend into obj[key], if present
        if key in obj:
            return get_value_by_path(obj[key], rest)
        else:
            return None
    elif isinstance(obj, list):
        # If current object is a list, we apply the search to each item
        results = []
        for item in obj:
            val = get_value_by_path(item, path_segments)
            if val is not None:
                # If the recursion returns a list, flatten it; otherwise just append
                if isinstance(val, list):
                    results.extend(val)
                else:
                    results.append(val)
        return results if results else None
    else:
        # We've hit a leaf (string, number, etc.) but still have path segments -> no match
        return None

def build_search_nodes(comp_nodes, search_mapping):
    """
    Build the 'sn' array using the search_mapping. For each mapping:
      - We find any flattened node from comp_nodes whose data "d" has an
        'archetype_node_id' (or 'ani') that matches the mapping's archetype_node_id.
      - We optionally include the 'a' array if createAncestorsArray == 'true'.
      - We collect in 'd' only the fields specified in 'data': [...paths...].
        Each path can traverse nested dict or list (e.g. /other_participations/performer/identifiers/id).
    """
    version = search_mapping.get("version", 1)
    mappings = search_mapping.get("mappings", [])
    result_nodes = []

    for mapping in mappings:
        desired_ai = mapping.get("archetype_node_id")
        # read whether to create the 'a' array
        create_ancestors = str(mapping.get("createAncestorsArray", "false")).lower() == "true"

        # For each node in our canonical flatten
        for node in comp_nodes:
            node_ai = node.get("d", {}).get("archetype_node_id") or node.get("d", {}).get("ani")
            if node_ai == desired_ai:
                # Start building the "search_node" object
                out_node = {}
                
                # Conditionally add ancestors
                if create_ancestors:
                    out_node["a"] = node.get("a", [])

                # Build data portion "d"
                d_out = {}
                # For each path in mapping["data"], locate it in node["d"]
                for attr_path in mapping.get("data", []):
                    # For example: attr_path might be "/_type" or "/time" or "/items[at0007]/items[at0014]/value/defining_code/code_string"
                    path_segments = attr_path.strip("/").split("/")
                    found_value = get_value_by_path(node["d"], path_segments)

                    if found_value is not None:
                        # We store it under the final segment name, e.g. 'code_string'
                        last_key = path_segments[-1]
                        d_out[last_key] = found_value

                out_node["d"] = d_out
                result_nodes.append(out_node)

    return {"version": version, "nodes": result_nodes}

def generate_synthetic_groups_by_ehr(processed_docs, min_comps, num_patients):
    """
    Groups processed docs by their EHR, keeps only those with at least 'min_comps',
    then replicates those groups until we have 'num_patients' groups.
    Assigns new EHR IDs each time so we have unique EHR groups.
    """
    groups_by_ehr = {}
    for doc in processed_docs:
        orig_ehr = doc["original"].get("ehr_id")
        if not orig_ehr:
            orig_ehr = generate_unique_id()
        groups_by_ehr.setdefault(orig_ehr, []).append(doc)

    unique_groups = [group for group in groups_by_ehr.values() if len(group) >= min_comps]
    logging.info(f"Found {len(groups_by_ehr)} patient groups; "
                 f"{len(unique_groups)} remain after filtering by min_compositions={min_comps}.")

    synthetic_groups = copy.deepcopy(unique_groups)
    while len(synthetic_groups) < num_patients:
        for grp in unique_groups:
            if len(synthetic_groups) >= num_patients:
                break
            synthetic_groups.append(copy.deepcopy(grp))

    synthetic_groups = synthetic_groups[:num_patients]
    for idx, group in enumerate(synthetic_groups):
        new_ehr = generate_unique_id()
        for doc in group:
            doc["ehr_id"] = new_ehr
        logging.info(f"Assigned synthetic ehr_id {new_ehr} to group {idx+1}/{len(synthetic_groups)}")
    return synthetic_groups

def assign_synthetic_composition_ids(synthetic_groups):
    """
    For each doc in each group, assign a new random composition id to ensure uniqueness.
    """
    for group in synthetic_groups:
        for doc in group:
            doc["synth_comp_id"] = generate_unique_id()

def process_all_documents(connection_string, config):
    global CONFIG_SHORTCUTS_ENABLED, CONFIG_SHORTCUTS
    CONFIG_SHORTCUTS_ENABLED = config.get("shortcuts_enabled", False)
    CONFIG_SHORTCUTS = config.get("shortcuts", {})

    logging.info(f"Shortcuts enabled: {CONFIG_SHORTCUTS_ENABLED}")
    logging.info(f"Shortcuts dictionary: {CONFIG_SHORTCUTS}")

    source_db_name = config["source_database_name"]
    source_col_name = config["source_collection"]
    target_col_name = config["target_collection"]
    batch_size = config.get("batch_size", 1000)
    doc_limit = config.get("limit", None)
    min_comps = config.get("min_compositions", 3)
    num_patients = config.get("num_patients", 30000)
    clean = config.get("clean_collections", True)
    
    # Template-specific search_nodes mappings
    search_nodes_mapping = config.get("search_nodes", {})

    logging.info("Connecting to MongoDB...")
    client = MongoClient(connection_string, tlsCAFile=certifi.where())
    db = client[source_db_name]
    src = db[source_col_name]
    target_collection = db[target_col_name]

    # 1) (Optional) Clean target collection
    if clean:
        logging.info("Clearing target collection...")
        target_collection.delete_many({})
    else:
        logging.info("Keeping existing data; new documents will be added.")

    # 2) Read all sample compositions, flatten, parse => store
    processed_docs = []
    count = 0
    t0 = time.time()
    for doc in src.find({}):
        processed_docs.append(process_document(doc))
        count += 1
        if count % 1000 == 0:
            logging.info(f"Processed {count} documents from the original samples...")
        if doc_limit and count >= doc_limit:
            break
    logging.info(f"Finished processing {count} documents in {time.time()-t0:.2f} seconds.")

    # 3) Generate synthetic docs from processed docs
    synthetic_groups = generate_synthetic_groups_by_ehr(processed_docs, min_comps, num_patients)
    assign_synthetic_composition_ids(synthetic_groups)

    # 4) Build final docs and insert them in batches
    bulk_docs = []
    total_inserted = 0

    # We'll track which template IDs are missing mappings so we don't spam logs.
    missing_mappings_recorded = set()

    for group in synthetic_groups:
        for doc in group:
            template_id = doc.get("template_id")
            if template_id and template_id in search_nodes_mapping:
                mapping_config = search_nodes_mapping[template_id]
                sn_obj = build_search_nodes(doc["nodes"], mapping_config)
                sn_nodes = sn_obj["nodes"]
                sn_version = mapping_config.get("version", 1)
            else:
                # If there's no mapping, we won't add the "sn" field at all.
                if template_id and template_id not in missing_mappings_recorded:
                    logging.warning(f"No search_nodes mapping found for template: {template_id}")
                    missing_mappings_recorded.add(template_id)
                sn_nodes = []
                sn_version = None

            hybrid_doc = {
                "_id": doc["synth_comp_id"],   # Unique ID for each synthetic doc
                "ehr_id": doc["ehr_id"],
                "cn": doc["nodes"]
            }
            # Add sn only if it's not empty
            if sn_nodes:
                hybrid_doc["sn"] = sn_nodes
            if sn_version and sn_nodes:
                hybrid_doc["snv"] = sn_version

            # 5) Apply shortcuts at the END if enabled
            if CONFIG_SHORTCUTS_ENABLED:
                hybrid_doc = apply_field_shortcuts(hybrid_doc)

            bulk_docs.append(InsertOne(hybrid_doc))
            total_inserted += 1

            # Insert in batches
            if len(bulk_docs) >= batch_size:
                target_collection.bulk_write(bulk_docs)
                logging.info(f"Inserted batch of {len(bulk_docs)} documents. Total processed: {total_inserted}")
                bulk_docs = []

    # Insert any remaining docs
    if bulk_docs:
        target_collection.bulk_write(bulk_docs)
        logging.info(f"Inserted final batch of {len(bulk_docs)} documents. Total processed: {total_inserted}")

    upload_time = time.time() - t0
    logging.info(f"All documents have been inserted. Total upload time: {upload_time:.2f} seconds.")
    client.close()
    logging.info("MongoDB connection closed.")

import { parseFromClause } from './hybrid-nodes/from-parser';
import { parseContainsClauses } from './hybrid-nodes/contains-parser';
import { parseWhereConditions } from './hybrid-nodes/where-parser';
import { buildMatchStage } from './hybrid-nodes/match-builder';
import { buildSearchStage } from './hybrid-nodes/search-builder';
import { buildAddFieldsStage } from './hybrid-nodes/addfields-builder';
import { buildProjectStage } from './hybrid-nodes/project-builder';
import { buildOptionsStages } from './hybrid-nodes/options-builder';

/**
 * Hybrid-Nodes Strategy for AQL to MQL Translation
 * This strategy intelligently chooses between $match and $search
 * based on the type of query (ehr_id-based or population-based).
 * 
 * @param {Object} ast - The AQL Abstract Syntax Tree
 * @param {Object} strategyConfig - Configuration for the Hybrid-Nodes strategy
 * @returns {Array} MongoDB aggregation pipeline
 */
export function translateHybridNodes(ast, strategyConfig = {}) {
  // Validate input
  if (!ast) {
    throw new Error("[Hybrid-Nodes] Invalid AST: No AST data provided");
  }
  
  console.log("[Hybrid-Nodes] Starting translation with config:", strategyConfig);
  
  // Step 1: Parse the FROM clause to extract ehr_id and determine query type
  const fromInfo = parseFromClause(ast);
  console.log("[Hybrid-Nodes] Parsed FROM clause:", fromInfo);
  
  // Step 2: Parse the CONTAINS clauses to extract archetype requirements
  const containsInfo = parseContainsClauses(ast);
  console.log("[Hybrid-Nodes] Parsed CONTAINS clauses, found", 
              containsInfo.entries.length, "entries");
  
  // Step 3: Parse the WHERE conditions
  const whereInfo = parseWhereConditions(ast, containsInfo);
  console.log("[Hybrid-Nodes] Parsed WHERE conditions, found", 
              whereInfo.conditions.length, "conditions");
  
  // --- Fix incomplete where conditions if needed ---
  // This is a special handling for ASTs that might have incomplete conditions
  if (whereInfo.conditions) {
    // Check for any conditions that are missing operators or values
    const incompleteConditions = whereInfo.conditions.filter(
      cond => !cond.operator || cond.value === undefined
    );
    
    if (incompleteConditions.length > 0) {
      console.warn([Hybrid-Nodes] Detected ${incompleteConditions.length} incomplete WHERE conditions. Fixing...);
      
      // Apply fixes to incomplete conditions
      incompleteConditions.forEach((cond, i) => {
        if (!cond.operator) {
          console.warn([Hybrid-Nodes] Fixing incomplete condition at index ${i}: Missing operator.);
          // Default to equals operator
          cond.operator = '=';
          cond.mongoOperator = '$eq';
        }
        
        if (cond.value === undefined) {
          console.warn([Hybrid-Nodes] Fixing incomplete condition at index ${i}: Missing value.);
          // Default to null value
          cond.value = null;
          cond.mongoValue = null;
        }
      });
      
      console.log("[Hybrid-Nodes] Updated WHERE conditions after fixing:", 
                 JSON.stringify(whereInfo.conditions, null, 2));
    }
  }
  
  // Initialize the pipeline
  const pipeline = [];
  
  // Step 4: Build the filtering stage ($match or $search) based on query type
  if (fromInfo.isEhrIdBased || !isAtlasSearchEnabled(strategyConfig)) {
    // For ehr_id-based queries or when Atlas Search is disabled, use $match
    console.log("[Hybrid-Nodes] Using $match with comp_nodes for filtering");
    const matchStage = buildMatchStage(fromInfo, containsInfo, whereInfo);
    if (matchStage && Object.keys(matchStage.$match || {}).length > 0) {
      console.log("[Hybrid-Nodes] $match stage created:", JSON.stringify(matchStage, null, 2));
      pipeline.push(matchStage);
    } else {
      console.warn("[Hybrid-Nodes] $match stage was empty. Check WHERE clause processing.");
    }
  } else {
    // For population-based queries with Atlas Search enabled, use $search
    console.log("[Hybrid-Nodes] Using Atlas $search with search_nodes for filtering");
    const searchStage = buildSearchStage(containsInfo, whereInfo, strategyConfig);
    if (searchStage) {
      console.log("[Hybrid-Nodes] $search stage created:", JSON.stringify(searchStage, null, 2));
      pipeline.push(searchStage);
      
      // Add a post-search $match stage if needed for conditions not handled by $search
      if (whereInfo.unhandledConditions && whereInfo.unhandledConditions.length > 0) {
        const postSearchMatchStage = buildPostSearchMatchStage(whereInfo.unhandledConditions);
        if (postSearchMatchStage) {
          console.log("[Hybrid-Nodes] Post-search $match stage added:", 
                     JSON.stringify(postSearchMatchStage, null, 2));
          pipeline.push(postSearchMatchStage);
        }
      } else {
        console.log("[Hybrid-Nodes] No additional post-search conditions found.");
      }
    } else {
      // Fall back to $match if $search couldn't be built
      console.warn("[Hybrid-Nodes] Could not build $search stage, falling back to $match");
      const matchStage = buildMatchStage(fromInfo, containsInfo, whereInfo);
      if (matchStage && Object.keys(matchStage.$match || {}).length > 0) {
        console.log("[Hybrid-Nodes] Fallback $match stage created:", 
                   JSON.stringify(matchStage, null, 2));
        pipeline.push(matchStage);
      } else {
        console.warn("[Hybrid-Nodes] Fallback $match stage was empty. Check WHERE clause processing.");
      }
    }
  }
  
  // Step 5: Build the $addFields stage to extract referenced entities
  const nodeField = "cn"; // Always use comp_nodes for entity extraction
  console.log([Hybrid-Nodes] Using ${nodeField} for entity extraction);
  
  const addFieldsStage = buildAddFieldsStage(ast, containsInfo, nodeField);
  if (addFieldsStage && Object.keys(addFieldsStage).length > 0) {
    pipeline.push({ $addFields: addFieldsStage });
  } else {
    console.warn("[Hybrid-Nodes] $addFields stage is empty.");
  }
  
  // Step 6: Build the $project stage for the SELECT clause
  const projectStage = buildProjectStage(ast, containsInfo);
  if (projectStage && Object.keys(projectStage).length > 0) {
    pipeline.push({ $project: projectStage });
  } else {
    console.warn("[Hybrid-Nodes] $project stage is empty.");
  }
  
  // Step 7: Build the remaining pipeline stages (sort, limit, etc.)
  const optionsStages = buildOptionsStages(ast);
  if (optionsStages && optionsStages.length > 0) {
    pipeline.push(...optionsStages);
  }
  
  console.log("[Hybrid-Nodes] Translation complete, pipeline stages count:", pipeline.length);
  return pipeline;
}

/**
 * Build a post-search $match stage for conditions not handled by $search.
 * 
 * @param {Array} unhandledConditions - Conditions not handled by Atlas Search
 * @returns {Object} MongoDB $match stage
 */
function buildPostSearchMatchStage(unhandledConditions) {
  if (!unhandledConditions || unhandledConditions.length === 0) {
    console.log("[Hybrid-Nodes] No unhandled WHERE conditions for post-search $match");
    return null;
  }
  
  const matchStage = { $match: {} };
  
  // Handle multiple conditions
  if (unhandledConditions.length > 1) {
    matchStage.$match.$and = unhandledConditions.map(cond => {
      const result = {};
      // Convert each condition to MongoDB query syntax
      if (cond.path && cond.operator) {
        const fieldName = cond.path.replace(/\//g, '_').replace(/[\[\]]/g, '_');
        
        // Handle various operators
        switch (cond.operator) {
          case '=':
            result[fieldName] = cond.mongoValue;
            break;
          case '!=':
            result[fieldName] = { $ne: cond.mongoValue };
            break;
          case '>':
            result[fieldName] = { $gt: cond.mongoValue };
            break;
          case '>=':
            result[fieldName] = { $gte: cond.mongoValue };
            break;
          case '<':
            result[fieldName] = { $lt: cond.mongoValue };
            break;
          case '<=':
            result[fieldName] = { $lte: cond.mongoValue };
            break;
          case 'MATCHES':
            // Handle MATCHES operation
            if (Array.isArray(cond.mongoValue)) {
              result[fieldName] = { $in: cond.mongoValue };
            } else {
              // Convert to regex for pattern matching
              const pattern = typeof cond.mongoValue === 'string' ? 
                cond.mongoValue.replace(/^'|'$/g, '').replace(/%/g, '.*').replace(/_/g, '.') : 
                cond.mongoValue;
              result[fieldName] = { $regex: pattern };
            }
            break;
          default:
            // Default to equality
            result[fieldName] = cond.mongoValue;
        }
      }
      return result;
    });
  }
  // Handle single condition
  else {
    const cond = unhandledConditions[0];
    if (cond.path && cond.operator) {
      const fieldName = cond.path.replace(/\//g, '_').replace(/[\[\]]/g, '_');
      
      // Handle various operators
      switch (cond.operator) {
        case '=':
          matchStage.$match[fieldName] = cond.mongoValue;
          break;
        case '!=':
          matchStage.$match[fieldName] = { $ne: cond.mongoValue };
          break;
        case '>':
          matchStage.$match[fieldName] = { $gt: cond.mongoValue };
          break;
        case '>=':
          matchStage.$match[fieldName] = { $gte: cond.mongoValue };
          break;
        case '<':
          matchStage.$match[fieldName] = { $lt: cond.mongoValue };
          break;
        case '<=':
          matchStage.$match[fieldName] = { $lte: cond.mongoValue };
          break;
        case 'MATCHES':
          // Handle MATCHES operation
          if (Array.isArray(cond.mongoValue)) {
            matchStage.$match[fieldName] = { $in: cond.mongoValue };
          } else {
            // Convert to regex for pattern matching
            const pattern = typeof cond.mongoValue === 'string' ? 
              cond.mongoValue.replace(/^'|'$/g, '').replace(/%/g, '.*').replace(/_/g, '.') : 
              cond.mongoValue;
            matchStage.$match[fieldName] = { $regex: pattern };
          }
          break;
        default:
          // Default to equality
          matchStage.$match[fieldName] = cond.mongoValue;
      }
    }
  }
  
  return matchStage;
}

/**
 * Parse the FROM clause of an AQL query to extract ehr_id and determine query type.
 * 
 * @param {Object} ast - The AQL Abstract Syntax Tree
 * @returns {Object} Information from the FROM clause
 */
export function parseFromClause(ast) {
  // Default return object
  const fromInfo = {
    isEhrIdBased: false,
    ehrId: null,
    ehrAlias: 'e', // Default EHR alias
    containmentStructure: {}
  };
  
  // Return default if there's no FROM clause
  if (!ast || !ast.from) {
    return fromInfo;
  }
  
  try {
    // Extract the EHR alias
    if (ast.from.alias) {
      fromInfo.ehrAlias = ast.from.alias;
    } else if (ast.from.rmType === 'EHR') {
      fromInfo.ehrAlias = ast.from.alias || 'e';
    }
    
    // Check if there's an EHR ID predicate in the FROM clause
    if (ast.from.predicate) {
      if (ast.from.predicate.path === 'ehr_id/value') {
        fromInfo.isEhrIdBased = true;
        fromInfo.ehrId = processValueForMongo(ast.from.predicate.value, ast.from.predicate.operator);
      }
    }
    
    // If not found in FROM predicate, check WHERE clause for ehr_id conditions
    if (!fromInfo.isEhrIdBased && ast.where) {
      // Handle different WHERE clause formats
      if (ast.where.path === ${fromInfo.ehrAlias}/ehr_id/value || ast.where.path === 'e/ehr_id/value') {
        fromInfo.isEhrIdBased = true;
        fromInfo.ehrId = processValueForMongo(ast.where.value, ast.where.operator);
      } 
      // Check if there's an AND condition with ehr_id
      else if (ast.where.operator === 'AND' && ast.where.conditions) {
        // Check each condition in the AND block
        Object.values(ast.where.conditions).forEach(condition => {
          if (condition.path === ${fromInfo.ehrAlias}/ehr_id/value || condition.path === 'e/ehr_id/value') {
            fromInfo.isEhrIdBased = true;
            fromInfo.ehrId = processValueForMongo(condition.value, condition.operator);
          }
        });
      }
    }
    
    // Build containment structure (for complex FROM clauses with multiple levels)
    buildContainmentStructure(ast.from, fromInfo.containmentStructure);
    
  } catch (error) {
    console.error("[FROM Parser] Error parsing FROM clause:", error);
  }
  
  return fromInfo;
}

/**
 * Build a containment structure from the FROM clause.
 * This helps track the relationship between different entities in the query.
 * 
 * @param {Object} fromClause - The FROM clause from the AST
 * @param {Object} containmentStructure - Object to populate with the structure
 */
function buildContainmentStructure(fromClause, containmentStructure) {
  if (!fromClause) return;
  
  // Get the root entity type and alias
  const rootType = fromClause.rmType || 'EHR';
  const rootAlias = fromClause.alias || 'e';
  
  // Set the root node
  containmentStructure[rootAlias] = {
    type: rootType,
    archetype: null,
    children: {}
  };
  
  // Process CONTAINS clauses if they exist
  if (fromClause.contains) {
    processContains(fromClause.contains, rootAlias, containmentStructure);
  }
}

/**
 * Process a CONTAINS clause and add it to the containment structure.
 * 
 * @param {Object} contains - The CONTAINS clause from the AST
 * @param {string} parentAlias - The alias of the parent entity
 * @param {Object} containmentStructure - Structure being built
 */
function processContains(contains, parentAlias, containmentStructure) {
  if (!contains) return;
  
  // Check if this is a simple CONTAINS or an AND/OR combination
  if (contains.rmType) {
    // Simple CONTAINS with a single entity
    const entityType = contains.rmType;
    const entityAlias = contains.alias || '';
    const archetype = contains.predicate && contains.predicate.path === 'archetype_node_id' 
      ? contains.predicate.value 
      : null;
    
    // Add to the parent's children
    if (containmentStructure[parentAlias] && entityAlias) {
      containmentStructure[parentAlias].children[entityAlias] = {
        type: entityType,
        archetype: archetype,
        children: {}
      };
      
      // If this entity contains others, process them
      if (contains.contains) {
        processContains(contains.contains, entityAlias, containmentStructure);
      }
    }
  } 
  // Handle AND/OR combinations
  else if (contains.operator === 'AND' && contains.children) {
    // Process each child in the AND combination
    Object.values(contains.children).forEach(child => {
      if (child.rmType) {
        const entityType = child.rmType;
        const entityAlias = child.alias || '';
        const archetype = child.predicate && child.predicate.path === 'archetype_node_id' 
          ? child.predicate.value 
          : null;
        
        // Add to the parent's children
        if (containmentStructure[parentAlias] && entityAlias) {
          containmentStructure[parentAlias].children[entityAlias] = {
            type: entityType,
            archetype: archetype,
            children: {}
          };
          
          // If this entity contains others, process them
          if (child.contains) {
            processContains(child.contains, entityAlias, containmentStructure);
          }
        }
      }
    });
  }
}

export default parseFromClause;

// lib/aqlToMql/translator/hybrid-nodes/contains-parser.js

/**
 * Parse the CONTAINS clauses of an AQL query to extract archetype requirements.
 * This parser handles different formats of CONTAINS clauses, including nested structures.
 * 
 * @param {Object} ast - The AQL Abstract Syntax Tree
 * @returns {Object} Information from the CONTAINS clauses
 */
 export function parseContainsClauses(ast) {
  // Create the default containsInfo structure
  const containsInfo = {
    entries: [],            // Array of all entries (flat list)
    entriesByAlias: {},     // Entries indexed by alias for quick lookup
    containmentTree: {},    // Hierarchical representation of the containment structure
    hasMultipleArchetypes: false // Flag to indicate if there are multiple archetypes
  };
  
  // Return default if no AST or CONTAINS clause
  if (!ast || !ast.contains) {
    return containsInfo;
  }
  
  try {
    // Process the CONTAINS clause(s)
    processContains(ast.contains, null, containsInfo);
    
    // Set the hasMultipleArchetypes flag if more than one archetype is specified
    containsInfo.hasMultipleArchetypes = containsInfo.entries.length > 1;
    
    // Index entries by alias for quick lookup
    containsInfo.entries.forEach(entry => {
      if (entry.alias) {
        containsInfo.entriesByAlias[entry.alias] = entry;
      }
    });
    
    // Build containment tree based on parent-child relationships
    buildContainmentTree(containsInfo);
    
  } catch (error) {
    console.error("[CONTAINS Parser] Error parsing CONTAINS clauses:", error);
  }
  
  return containsInfo;
}

/**
 * Process a CONTAINS clause and extract its information.
 * 
 * @param {Object} contains - CONTAINS clause from the AST
 * @param {string|null} parentAlias - Alias of the parent entity, or null for top-level
 * @param {Object} containsInfo - Object being populated with CONTAINS information
 */
function processContains(contains, parentAlias, containsInfo) {
  // Handle different formats of CONTAINS
  if (!contains) return;
  
  // Handle AND operator with multiple CONTAINS clauses
  if (contains.operator === 'AND' && contains.children) {
    // Process each child with the same parent
    Object.values(contains.children).forEach(child => {
      processContains(child, parentAlias, containsInfo);
    });
    return;
  }
  
  // Handle simple CONTAINS with a single entity
  if (contains.rmType) {
    // Extract archetype ID from predicate if present
    let archetypeId = null;
    if (contains.predicate && contains.predicate.path === 'archetype_node_id') {
      archetypeId = contains.predicate.value;
    }
    
    // Create an entry for this entity
    const entry = {
      alias: contains.alias || '',
      rmType: contains.rmType,
      archetypeId: archetypeId,
      parentAlias: parentAlias,
      path: contains.path || '',
      dataPaths: [], // Will be populated during WHERE or SELECT parsing
      constraints: [] // Additional constraints if needed
    };
    
    // Add to entries array
    containsInfo.entries.push(entry);
    
    // Process nested CONTAINS if present
    if (contains.contains) {
      processContains(contains.contains, entry.alias, containsInfo);
    }
  }
  // Handle OR operator (just process each side independently)
  else if (contains.operator === 'OR' && contains.children) {
    // Process each child with the same parent
    Object.values(contains.children).forEach(child => {
      processContains(child, parentAlias, containsInfo);
    });
  }
}

/**
 * Build a tree structure representing the containment hierarchy.
 * 
 * @param {Object} containsInfo - CONTAINS information with entries
 */
function buildContainmentTree(containsInfo) {
  // First, organize entries by their parent
  const entriesByParent = {};
  
  // Find all top-level entries (those without a parent)
  const topLevelEntries = [];
  
  // Categorize entries by their parent
  containsInfo.entries.forEach(entry => {
    if (!entry.parentAlias) {
      // This is a top-level entry
      topLevelEntries.push(entry);
    } else {
      // This is a child entry, group by parent alias
      if (!entriesByParent[entry.parentAlias]) {
        entriesByParent[entry.parentAlias] = [];
      }
      entriesByParent[entry.parentAlias].push(entry);
    }
  });
  
  // Build the tree recursively
  function buildSubtree(entries) {
    const tree = {};
    
    entries.forEach(entry => {
      // Create a node for this entry
      tree[entry.alias] = {
        rmType: entry.rmType,
        archetypeId: entry.archetypeId,
        children: {}
      };
      
      // Add children if this entry has any
      const children = entriesByParent[entry.alias] || [];
      if (children.length > 0) {
        tree[entry.alias].children = buildSubtree(children);
      }
    });
    
    return tree;
  }
  
  // Start building the tree from top-level entries
  containsInfo.containmentTree = buildSubtree(topLevelEntries);
}

/**
 * Helper function to get containment path for an archetype
 * Returns an array of ancestors from root to the given archetype
 * 
 * @param {string} alias - The alias to find the path for
 * @param {Object} containsInfo - Contains information
 * @returns {Array} Array of aliases forming the path
 */
export function getContainmentPath(alias, containsInfo) {
  const path = [];
  
  // Helper function to find the path recursively
  function findPath(currentAlias) {
    // Add this alias to the path
    path.unshift(currentAlias);
    
    // Get the entry for this alias
    const entry = containsInfo.entriesByAlias[currentAlias];
    
    // If it has a parent, add the parent's path
    if (entry && entry.parentAlias) {
      findPath(entry.parentAlias);
    }
  }
  
  // Start the search with the given alias
  findPath(alias);
  
  return path;
}

/**
 * Helper function to find entry by archetype ID
 * 
 * @param {string} archetypeId - The archetype ID to look for
 * @param {Object} containsInfo - Contains information
 * @returns {Object|null} The entry with the matching archetype ID, or null if not found
 */
export function findEntryByArchetypeId(archetypeId, containsInfo) {
  for (const entry of containsInfo.entries) {
    if (entry.archetypeId === archetypeId) {
      return entry;
    }
  }
  return null;
}

/**
 * Parse the WHERE clause of an AQL query to extract conditions.
 * 
 * @param {Object} ast - The AQL Abstract Syntax Tree
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @returns {Object} Information from the WHERE conditions
 */
export function parseWhereConditions(ast, containsInfo) {
  // Create default structure
  const whereInfo = {
    conditions: [],             // All conditions (flat list)
    ehrIdCondition: null,       // The ehr_id condition if present
    archetypeConditions: [],    // Conditions on archetype properties
    pathConditions: {},         // Conditions organized by path
    valueConditions: [],        // Conditions on values
    timeConditions: [],         // Conditions on timestamps
    logicalGroups: [],          // AND/OR logical groups
    unhandledConditions: []     // Conditions that couldn't be processed
  };
  
  // Return default if no WHERE clause
  if (!ast || !ast.where) {
    return whereInfo;
  }
  
  try {
    // Process the WHERE clause based on its structure
    if (ast.where.path && ast.where.operator) {
      // Single condition
      processSingleCondition(ast.where, whereInfo, containsInfo);
    } else if (ast.where.operator && ast.where.conditions) {
      // Logical group (AND/OR)
      processLogicalGroup(ast.where, whereInfo, containsInfo);
    }
    
    // Check for incomplete conditions (missing operator/value)
    fixIncompleteConditions(whereInfo);
    
  } catch (error) {
    console.error("[WHERE Parser] Error parsing WHERE conditions:", error);
  }
  
  return whereInfo;
}

/**
 * Process a single WHERE condition.
 * 
 * @param {Object} condition - The WHERE condition from the AST
 * @param {Object} whereInfo - Object being populated with WHERE information
 * @param {Object} containsInfo - Information from the CONTAINS clause
 */
function processSingleCondition(condition, whereInfo, containsInfo) {
  // Parse the path to extract alias and path components
  const { alias, path } = parseAqlPath(condition.path);
  
  // Create parsed condition with all necessary properties
  const parsedCondition = {
    path: condition.path,
    pathComponents: { alias, path },
    operator: condition.operator,
    mongoOperator: mapOperatorToMongo(condition.operator),
    value: condition.value,
    mongoValue: processValueForMongo(condition.value, condition.operator),
    archetypeIds: extractArchetypeIdsFromPath(path || ''),
    isEhrIdCondition: false,
    isArchetypeCondition: false,
    isTimeCondition: false,
    isValueCondition: false
  };
  
  // Determine the type of condition
  if (condition.path.includes('ehr_id/value')) {
    parsedCondition.isEhrIdCondition = true;
    whereInfo.ehrIdCondition = parsedCondition;
  } else if (condition.path.includes('archetype_node_id') || condition.path.includes('archetype_details')) {
    parsedCondition.isArchetypeCondition = true;
    whereInfo.archetypeConditions.push(parsedCondition);
  } else if (condition.path.includes('/time/value') || 
             condition.path.includes('start_time') || 
             condition.path.includes('date')) {
    parsedCondition.isTimeCondition = true;
    whereInfo.timeConditions.push(parsedCondition);
  } else if (condition.path.includes('/value/value') || 
             condition.path.includes('/value/magnitude') || 
             condition.path.includes('/value/defining_code/code_string')) {
    parsedCondition.isValueCondition = true;
    whereInfo.valueConditions.push(parsedCondition);
  } else {
    whereInfo.unhandledConditions.push(parsedCondition);
  }
  
  // Add to main conditions list
  whereInfo.conditions.push(parsedCondition);
  
  // Add to pathConditions for path-based lookup
  if (!whereInfo.pathConditions[condition.path]) {
    whereInfo.pathConditions[condition.path] = [];
  }
  whereInfo.pathConditions[condition.path].push(parsedCondition);
}

/**
 * Process a logical group (AND/OR) of conditions.
 * 
 * @param {Object} group - The logical group from the AST
 * @param {Object} whereInfo - Object being populated with WHERE information
 * @param {Object} containsInfo - Information from the CONTAINS clause
 */
function processLogicalGroup(group, whereInfo, containsInfo) {
  const logicalGroup = {
    operator: group.operator,
    mongoOperator: group.operator === 'AND' ? '$and' : '$or',
    conditions: []
  };
  
  // Process each condition in the group
  Object.values(group.conditions).forEach(condition => {
    if (condition.path && condition.operator) {
      // Single condition within the group
      processSingleCondition(condition, whereInfo, containsInfo);
      
      // Add to the logical group's conditions
      logicalGroup.conditions.push(condition);
    } else if (condition.operator && condition.conditions) {
      // Nested logical group
      const nestedGroup = {
        operator: condition.operator,
        mongoOperator: condition.operator === 'AND' ? '$and' : '$or',
        conditions: []
      };
      
      // Process the nested group recursively
      processLogicalGroup(condition, whereInfo, containsInfo);
      
      // Add to the parent logical group
      logicalGroup.conditions.push(nestedGroup);
    }
  });
  
  // Add to logical groups if not empty
  if (logicalGroup.conditions.length > 0) {
    whereInfo.logicalGroups.push(logicalGroup);
  }
}

/**
 * Fix incomplete conditions by filling in missing information.
 * This is helpful for handling potentially incomplete AST data.
 * 
 * @param {Object} whereInfo - Object with WHERE information to be fixed
 */
function fixIncompleteConditions(whereInfo) {
  // Find conditions missing operator or value
  whereInfo.conditions.forEach((condition, index) => {
    // Check if operator is missing
    if (!condition.operator || condition.operator === null) {
      console.warn([WHERE Parser] Condition at index ${index} is missing operator, applying default '=');
      condition.operator = '=';
      condition.mongoOperator = '$eq';
    }
    
    // Check if value is missing or null
    if (condition.value === null || condition.value === undefined) {
      console.warn([WHERE Parser] Condition at index ${index} is missing value, applying default null);
      condition.value = null;
      condition.mongoValue = null;
    }
    
    // Handle MATCHES with code_string - a common pattern in openEHR queries
    if (condition.operator === 'MATCHES' && 
        condition.path && 
        condition.path.endsWith('/value/defining_code/code_string')) {
      
      // Ensure MATCHES values are properly formatted as arrays
      if (!Array.isArray(condition.mongoValue)) {
        if (typeof condition.mongoValue === 'string') {
          // Handle the {'val1','val2'} format
          if (condition.mongoValue.startsWith('{') && condition.mongoValue.endsWith('}')) {
            const innerValue = condition.mongoValue.substring(1, condition.mongoValue.length - 1);
            condition.mongoValue = innerValue.split(',')
              .map(v => v.trim())
              .map(v => v.startsWith("'") && v.endsWith("'") ? v.substring(1, v.length - 1) : v);
          } else {
            condition.mongoValue = [condition.mongoValue];
          }
        } else {
          condition.mongoValue = [condition.mongoValue];
        }
      }
    }
  });
}

  export function mapArchetypePrefixToShorthand(archetypeId) {

      if (archetypeId.startsWith(prefix)) {
        return shorthand;
      }
    }
    return null;

/**
 * Convert a full archetype ID to its shorthand form
 * e.g., "openEHR-EHR-EVALUATION.adverse_reaction_risk.v2" -> "EE.adverse_reaction_risk.v2"
 * 
 * @param {string} archetypeId - Full archetype ID
 * @returns {string} Shorthand representation or original if not found
 */
 export function mapArchetypeIdToShorthand(archetypeId) {
    if (!archetypeId) return archetypeId;
    
    for (const [prefix, shorthand] of Object.entries(ARCHETYPE_PREFIX_MAP)) {
      if (archetypeId.startsWith(prefix)) {
        // Replace the prefix with the shorthand
        return archetypeId.replace(prefix, shorthand);
      }
    }
    
    return archetypeId;
  }

   * 
   * @param {string} dataType - Standard openEHR data type
   * @returns {string} Shorthand representation or original if not found

  export function mapDataTypeToShorthand(dataType) {
    return DATA_TYPE_MAP[dataType] || dataType;
  }
  
  /**
   * Convert a complete AQL path to a MongoDB path using appropriate shortcuts
   * This function handles nested shortcuts and field replacements
   * 
   * @param {string} aqlPath - AQL path (e.g., "ev/data[at0001]/items[at0002]/value/value")
   * @returns {string} Equivalent MongoDB path with shortcuts
   */
  export function mapAqlPathToMongoPath(aqlPath) {
    // First, handle archetype references in the path
    const pathWithArchetypes = transformArchetypeReferences(aqlPath);
    
    // Then replace field names with their shortcuts
    const segments = pathWithArchetypes.split('.');
    const transformedSegments = segments.map(segment => mapFieldNameToShorthand(segment));
    
    return transformedSegments.join('.');
  }
  
  /**
   * Transform archetype references in an AQL path
   * e.g., "data[at0001]/items[at0002]" to "data.at0001.items.at0002"
   * 
   * @param {string} aqlPath - AQL path with archetype references
   * @returns {string} Path with transformed archetype references
   */
  function transformArchetypeReferences(aqlPath) {
    // Replace [archetypeId] with .archetypeId
    let result = aqlPath.replace(/\[([^\]]+)\]/g, '.$1');
    
    // Replace forward slashes with dots
    result = result.replace(/\//g, '.');
    
    return result;
  }

   * Map an AQL operator to a MongoDB operator
   * 
   * @param {string} operator - AQL operator
   * @returns {string} MongoDB operator

  export function mapOperatorToMongo(operator) {
    const operatorMap = {
      '=': '$eq',
      '==': '$eq',
      '!=': '$ne',
      '>': '$gt',
      '>=': '$gte',
      '<': '$lt',
      '<=': '$lte',
      'EXISTS': '$exists',
      'MATCHES': '$in'
    };
    
    return operatorMap[operator] || '$eq';
  }

import { 
    mapFieldNameToShorthand, 
    mapArchetypePrefixToShorthand 
  } from './dictionary';
  
  /**
   * Parse an AQL path into components
   * 
   * @param {string} aqlPath - Full AQL path (e.g., "ev/data[at0001]/items[at0002]/value/value")
   * @returns {Object} Path components { alias, path, pathSegments, archetypeIds }
   */
  export function parseAqlPath(aqlPath) {
    if (!aqlPath) {
      return { alias: null, path: null, pathSegments: [], archetypeIds: [] };
    }
    
    // Split the path on the first slash to get the alias and the rest
    const firstSlashIndex = aqlPath.indexOf('/');
    
    if (firstSlashIndex === -1) {
      // No slash found, the whole thing is an alias
      return { 
        alias: aqlPath, 
        path: '', 
        pathSegments: [], 
        archetypeIds: [] 
      };
    }
    
    const alias = aqlPath.substring(0, firstSlashIndex);
    const path = aqlPath.substring(firstSlashIndex + 1);
    
    // Split path into segments and extract archetype IDs
    const pathSegments = [];
    const archetypeIds = [];
    
    // Parse the path segments and extract archetype IDs
    let currentSegment = '';
    let inBrackets = false;
    
    for (let i = 0; i < path.length; i++) {
      const char = path[i];
      
      if (char === '[') {
        // Start of an archetype ID
        inBrackets = true;
        // Push the segment name (e.g., "data" or "items")
        if (currentSegment) {
          pathSegments.push({ 
            type: 'field', 
            value: currentSegment 
          });
        }
        currentSegment = '';
      } else if (char === ']') {
        // End of an archetype ID
        inBrackets = false;
        // Push the archetype ID
        archetypeIds.push(currentSegment);
        pathSegments.push({ 
          type: 'archetype', 
          value: currentSegment 
        });
        currentSegment = '';
      } else if (char === '/' && !inBrackets) {
        // Path separator (only if not in brackets)
        if (currentSegment) {
          pathSegments.push({ 
            type: 'field', 
            value: currentSegment 
          });
        }
        currentSegment = '';
      } else {
        // Add to current segment
        currentSegment += char;
      }
    }
    
    // Add the last segment if there is one
    if (currentSegment) {
      pathSegments.push({ 
        type: 'field', 
        value: currentSegment 
      });
    }
    
    return { alias, path, pathSegments, archetypeIds };
  }
  
  /**
   * Convert an AQL path to a MongoDB path for the flattened structure
   * 
   * @param {string} alias - Path alias (e.g., "ev")
   * @param {string} path - AQL path after the alias (e.g., "data[at0001]/items[at0002]/value/value")
   * @returns {string} MongoDB path for the document structure
   */
  export function convertAqlPathToMongoPath(alias, path) {
    // Extract archetype segments and field segments
    const { pathSegments } = parseAqlPath(${alias}/${path});
    
    let result = '';
    
    // Handle simple paths without archetype IDs
    if (!path.includes('[')) {
      // Just convert slashes to dots
      result = path.replace(/\//g, '.');
      // Apply field name shortcuts
      const segments = result.split('.');
      result = segments.map(segment => mapFieldNameToShorthand(segment)).join('.');
      
      // Return prefixed by alias and d (data)
      return ${alias}.d.${result};
    }
    
    // Handle complex paths with archetype IDs
    let currentPath = '';
    let prefixParts = [alias, 'd'];
    
    for (const segment of pathSegments) {
      if (segment.type === 'field') {
        // Add field segment with shorthand mapping
        const shorthand = mapFieldNameToShorthand(segment.value);
        prefixParts.push(shorthand);
      }
      else if (segment.type === 'archetype') {
        // Check if it's a standard archetype ID
        const prefix = mapArchetypePrefixToShorthand(segment.value);
        if (prefix) {
          // Use the prefix for standard archetypes
          prefixParts.push(prefix);
        } else {
          // Use the archetype ID as is
          prefixParts.push(segment.value);
        }
      }
    }
    
    // Join all parts with dots
    return prefixParts.join('.');
  }
  
  /**
   * Build a variable name for a path that can be used in MongoDB aggregation pipeline
   * 
   * @param {string} alias - Path alias
   * @param {string} path - AQL path after the alias
   * @returns {string} Variable name safe for MongoDB
   */
  export function buildVariableNameForPath(alias, path) {
    // Replace characters that would be invalid in MongoDB variable names
    return ${alias}_${path.replace(/[\[\]\/.]/g, '_')};
  }
  
  /**
   * Extract archetype IDs from a path
   * e.g., "data[at0001]/items[openEHR-EHR-CLUSTER.medication.v2]/items[at0132]"
   * -> ["at0001", "openEHR-EHR-CLUSTER.medication.v2", "at0132"]
   * 
   * @param {string} path - AQL path with archetype references
   * @returns {Array} Array of archetype IDs
   */
   export function extractArchetypeIdsFromPath(path) {
    const archetypeIds = [];
    const matches = path.match(/\[(at\d+|openEHR-EHR-[A-Z_]+\.[a-z_]+\.v\d+)\]/g);
    
    if (matches) {
      matches.forEach(match => {
        const archetypeId = match.substring(1, match.length - 1);
        // Apply the shorthand mapping to transform full archetypes
        archetypeIds.push(mapArchetypeIdToShorthand(archetypeId));
      });
    }
    
    return archetypeIds;
  }
  
  /**
   * Process a value for MongoDB query, handling special cases
   * 
   * @param {*} value - Raw value from AQL query
   * @param {string} operator - AQL operator
   * @returns {*} Processed value for MongoDB
   */
  export function processValueForMongo(value, operator) {
    // Handle null/undefined values
    if (value === null || value === undefined) {
      return null;
    }
    
    // Handle MATCHES operator which can take an array or a string with format {'val1','val2'}
    if (operator === 'MATCHES' || operator === 'matches') {
      if (Array.isArray(value)) {
        return value;
      } else if (typeof value === 'string') {
        // Check for the {'val1','val2'} format
        if (value.startsWith('{') && value.endsWith('}')) {
          // Extract the values inside the braces
          const innerValue = value.substring(1, value.length - 1);
          // Split by comma, but handle quoted strings properly
          return innerValue.split(',')
            .map(v => v.trim())
            .map(v => v.startsWith("'") && v.endsWith("'") ? v.substring(1, v.length - 1) : v);
        } else {
          // Single value
          return [value];
        }
      } else if (value && value.type === 'literal' && value.value) {
        // Handle structured literal values from AST
        if (typeof value.value === 'string' && value.value.startsWith('{') && value.value.endsWith('}')) {
          // Extract the values inside the braces
          const innerValue = value.value.substring(1, value.value.length - 1);
          // Split by comma, but handle quoted strings properly
          return innerValue.split(',')
            .map(v => v.trim())
            .map(v => v.startsWith("'") && v.endsWith("'") ? v.substring(1, v.length - 1) : v);
        } else {
          return [value.value];
        }
      }
      
      // Default for other cases
      return [value];
    }
    
    // Handle parameter variables (e.g., $ehr_id)
    if (typeof value === 'string' && value.startsWith('$')) {
      return value;
    }
    
    // Handle date values
    if (value && (value.type === 'date' || (typeof value === 'object' && value.value && value.$date))) {
      return value.value || value.$date;
    }
    
    // Handle literal values from AST
    if (value && value.type === 'literal' && value.value !== undefined) {
      return value.value;
    }
    
    // Handle regular values
    return value;
  }
  
  export default {
    parseAqlPath,
    convertAqlPathToMongoPath,
    buildVariableNameForPath,
    extractArchetypeIdsFromPath,
    processValueForMongo
  };

/**
 * Build the MongoDB $match stage based on FROM, CONTAINS, and WHERE clauses.
 * 
 * @param {Object} fromInfo - Information from the parsed FROM clause
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @param {Object} whereInfo - Information from the parsed WHERE conditions
 * @returns {Object} MongoDB $match stage
 */
export function buildMatchStage(fromInfo, containsInfo, whereInfo) {
  // Create the base $match stage
  const matchStage = { $match: {} };
  
  // Add ehr_id condition if present
  if (fromInfo.isEhrIdBased && fromInfo.ehrId) {
    matchStage.$match.ehr_id = fromInfo.ehrId;
  }
  
  // Build an array of conditions
  const conditions = [];
  
  // Add archetype conditions from CONTAINS clauses
  const archetypeConditions = buildArchetypeConditions(containsInfo);
  if (archetypeConditions.length > 0) {
    conditions.push(...archetypeConditions);
  }
  
  // Add conditions from WHERE clause
  if (whereInfo && whereInfo.conditions.length > 0) {
    const whereConditions = buildWhereConditions(whereInfo);
    if (whereConditions.length > 0) {
      conditions.push(...whereConditions);
    }
  }
  
  // Only add conditions to $match if there are any
  if (conditions.length > 0) {
    // If we already have ehr_id in $match
    if (Object.keys(matchStage.$match).length > 0) {
      matchStage.$match.$and = conditions;
    } else {
      // No existing conditions, use $and directly
      matchStage.$match = { $and: conditions };
    }
  }
  
  return matchStage;
}

/**
 * Build MongoDB conditions from CONTAINS clauses.
 * 
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @returns {Array} Array of MongoDB conditions
 */
function buildArchetypeConditions(containsInfo) {
  const conditions = [];
  const nodesField = getNodesArrayField(false); // Use "cn" for comp_nodes
  
  // Add a condition for each required archetype
  containsInfo.entries.forEach(entry => {
    if (entry.archetypeId) {
      const condition = {
        [nodesField]: {
          $elemMatch: {
            "d.T": entry.rmType ? mapRmTypeToShorthand(entry.rmType) : { $exists: true }
          }
        }
      };
      
      const shortArchetypeId = mapArchetypeIdToShorthand(entry.archetypeId);
      condition[nodesField].$elemMatch.$or = [
        { "d.ani": shortArchetypeId },
        { "d.ad.ai.v": shortArchetypeId }
      ];
      
      // Add ancestor conditions if this is a nested archetype
      if (entry.parentAlias) {
        const parentEntry = containsInfo.entriesByAlias[entry.parentAlias];
        if (parentEntry && parentEntry.archetypeId) {
          condition[nodesField].$elemMatch.a = { $elemMatch: { $eq: parentEntry.archetypeId } };
        }
      }
      
      conditions.push(condition);
    }
  });
  
  return conditions;
}

/**
 * Build MongoDB conditions from WHERE clauses.
 * 
 * @param {Object} whereInfo - Information from the parsed WHERE conditions
 * @returns {Array} Array of MongoDB conditions
 */
function buildWhereConditions(whereInfo) {
  const conditions = [];
  const nodesField = getNodesArrayField(false); // Use "cn" for comp_nodes
  
  // Skip ehr_id conditions (handled separately)
  const nonEhrIdConditions = whereInfo.conditions.filter(cond => 
    !cond.isEhrIdCondition && cond.path && cond.operator);
  
  // Process each non-ehr_id condition
  nonEhrIdConditions.forEach(condition => {
    // Get the alias and path
    const { alias, path } = condition.pathComponents;
    
    // Handle different types of conditions
    if (condition.isValueCondition) {
      // Value condition (e.g., ev/data[at0001]/items[at0002]/value/value)
      conditions.push(buildValueCondition(condition, nodesField));
    } else if (condition.isTimeCondition) {
      // Time condition (e.g., ev/time/value)
      conditions.push(buildTimeCondition(condition, nodesField));
    } else if (condition.isArchetypeCondition) {
      // Archetype condition (e.g., ev/archetype_node_id)
      conditions.push(buildArchetypeNodeCondition(condition, nodesField));
    } else {
      // Generic condition based on path
      conditions.push(buildGenericCondition(condition, nodesField));
    }
  });
  
  return conditions;
}

/**
 * Build a MongoDB condition for value-related filters.
 * 
 * @param {Object} condition - The WHERE condition
 * @param {string} nodesField - Field name for the nodes array
 * @returns {Object} MongoDB condition
 */
function buildValueCondition(condition, nodesField) {
  const { alias, path } = condition.pathComponents;
  
  // Handle different patterns of value paths
  if (path.endsWith('/value/value')) {
    // Extract archetype IDs from the path for more precise matching
    const archetypeIds = condition.archetypeIds;
    
    if (archetypeIds.length > 0) {
      // Build a condition using archetype IDs
      return {
        [nodesField]: {
          $elemMatch: {
            "d.T": { $in: ["U", "E", "C"] }, // Common types for elements with values
            "a": archetypeIds.length > 0 ? { $all: archetypeIds } : { $exists: true },
            "d.v.v": { [condition.mongoOperator]: condition.mongoValue }
          }
        }
      };
    } else {
      // Simpler condition without archetype ID constraints
      return {
        [nodesField]: {
          $elemMatch: {
            "d.v.v": { [condition.mongoOperator]: condition.mongoValue }
          }
        }
      };
    }
  } 
  // Handle coded values (defining_code/code_string)
  else if (path.endsWith('/value/defining_code/code_string')) {
    // Extract archetype IDs from the path
    const archetypeIds = condition.archetypeIds;
    
    // Special handling for MATCHES operator with code_string
    if (condition.operator === 'MATCHES') {
      return {
        [nodesField]: {
          $elemMatch: {
            "d.T": { $in: ["U", "E", "C"] },
            "a": archetypeIds.length > 0 ? { $all: archetypeIds } : { $exists: true },
            "d.v.df.cs": { $in: condition.mongoValue }
          }
        }
      };
    } else {
      // Normal operator
      return {
        [nodesField]: {
          $elemMatch: {
            "d.T": { $in: ["U", "E", "C"] },
            "a": archetypeIds.length > 0 ? { $all: archetypeIds } : { $exists: true },
            "d.v.df.cs": { [condition.mongoOperator]: condition.mongoValue }
          }
        }
      };
    }
  } 
  // Default handling for other value paths
  else {
    return buildGenericCondition(condition, nodesField);
  }
}

/**
 * Build a MongoDB condition for time-related filters.
 * 
 * @param {Object} condition - The WHERE condition
 * @param {string} nodesField - Field name for the nodes array
 * @returns {Object} MongoDB condition
 */
function buildTimeCondition(condition, nodesField) {
  const { alias, path } = condition.pathComponents;
  
  // Handle ACTION.time/value
  if (path === 'time/value') {
    return {
      [nodesField]: {
        $elemMatch: {
          "d.T": "A", // ACTION
          "d.time.v": { [condition.mongoOperator]: condition.mongoValue }
        }
      }
    };

  // Handle COMPOSITION.context/start_time/value
  else if (path === 'context/start_time/value') {
    return {
      [nodesField]: {
        $elemMatch: {
          "d.T": "C", // COMPOSITION
          "d.cx.st.v": { [condition.mongoOperator]: condition.mongoValue }
        }
      }
    };
  } 
  // Default for other time paths
  else {
    return buildGenericCondition(condition, nodesField);
  }
}

/**
 * Build the MongoDB Atlas Search stage for population queries.
 * 
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @param {Object} whereInfo - Information from the parsed WHERE conditions
 * @param {Object} strategyConfig - Configuration options
 * @returns {Object} MongoDB $search stage or null if not applicable
 */
export function buildSearchStage(containsInfo, whereInfo, strategyConfig) {
  // Ensure that Atlas Search is properly configured
  if (!strategyConfig || !strategyConfig.atlasSearch || !strategyConfig.atlasSearch.index_name) {
    console.warn("[Search-Builder] Missing Atlas Search configuration. Cannot build $search stage.");
    return null;
  }

  // Create the $search stage
  const searchStage = {
    $search: {
      index: strategyConfig.atlasSearch.index_name,
      compound: {
        must: [],
        should: [],
        mustNot: []
      }
    }
  };

  // Add conditions for archetypes
  const archetypeConditions = buildArchetypeSearchConditions(containsInfo);
  if (archetypeConditions.length > 0) {
    searchStage.$search.compound.must.push(...archetypeConditions);
  }

  // Add conditions from WHERE clause
  if (whereInfo && whereInfo.conditions.length > 0) {
    // Identify which conditions can be handled by Atlas Search
    const searchableConditions = whereInfo.conditions.filter(cond =>
      !cond.isEhrIdCondition && isSearchableCondition(cond, strategyConfig));

    // Build search conditions for searchable WHERE conditions
    const whereSearchConditions = buildWhereSearchConditions(searchableConditions, strategyConfig);
    if (whereSearchConditions.length > 0) {
      searchStage.$search.compound.must.push(...whereSearchConditions);
    }

    // Identify conditions that cannot be handled by Atlas Search
    // These will need to be handled by a separate $match stage
    const unsearchableConditions = whereInfo.conditions.filter(cond =>
      !cond.isEhrIdCondition && !isSearchableCondition(cond, strategyConfig));

    // Store these for later processing
    if (unsearchableConditions.length > 0) {
      whereInfo.unhandledConditions = unsearchableConditions;
    }
  }

  // Only return the stage if it has at least one condition
  if (searchStage.$search.compound.must.length === 0 &&
    searchStage.$search.compound.should.length === 0 &&
    searchStage.$search.compound.mustNot.length === 0) {
    return null;
  }

  return searchStage;
}

/**
 * Build search conditions for archetypes required by CONTAINS clauses.
 * 
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @returns {Array} Array of Atlas Search conditions
 */
function buildArchetypeSearchConditions(containsInfo) {
  const conditions = [];

  // Add a condition for each required archetype
  containsInfo.entries.forEach(entry => {
    if (entry.archetypeId) {
      const searchCondition = {
        embeddedDocument: {
          path: "sn",
          operator: {
            compound: {
              must: [
                {
                  equals: {
                    path: "sn.d.ani",
                    value: entry.archetypeId
                  }
                }
              ]
            }
          }
        }
      };

      // Add RM type condition if available
      if (entry.rmType) {
        const shortArchetypeId = mapArchetypeIdToShorthand(entry.archetypeId);
        searchCondition.embeddedDocument.operator.compound.must.push({
          equals: {
            path: "sn.d.ani",
            value: shortArchetypeId
          }
        });
      }

      // Add parent relationship if this is a nested archetype
      if (entry.parentAlias) {
        const parentEntry = containsInfo.entriesByAlias[entry.parentAlias];
        if (parentEntry && parentEntry.archetypeId) {
          searchCondition.embeddedDocument.operator.compound.must.push({
            equals: {
              path: "sn.a",
              value: parentEntry.archetypeId
            }
          });
        }
      }

      conditions.push(searchCondition);
    }
  });

  return conditions;
}

/**
 * Build search conditions from WHERE clauses.
 * 
 * @param {Array} conditions - Searchable WHERE conditions
 * @param {Object} strategyConfig - Configuration options
 * @returns {Array} Array of Atlas Search conditions
 */
function buildWhereSearchConditions(conditions, strategyConfig) {
  const searchConditions = [];

  conditions.forEach(condition => {
    if (condition.isValueCondition) {
      // Value condition (e.g., ev/data[at0001]/items[at0002]/value/value)
      const valueCondition = buildValueSearchCondition(condition, strategyConfig);
      if (valueCondition) {
        searchConditions.push(valueCondition);
      }
    } else if (condition.isTimeCondition) {
      // Time condition (e.g., ev/time/value)
      const timeCondition = buildTimeSearchCondition(condition);
      if (timeCondition) {
        searchConditions.push(timeCondition);
      }
    } else if (condition.isArchetypeCondition) {
      // Archetype condition (e.g., ev/archetype_node_id)
      const archetypeCondition = buildArchetypeSearchCondition(condition);
      if (archetypeCondition) {
        searchConditions.push(archetypeCondition);
      }
    }
  });

  return searchConditions;
}

/**
 * Build a value search condition for value-related filters.
 * 
 * @param {Object} condition - The WHERE condition
 * @param {Object} strategyConfig - Configuration options
 * @returns {Object} Atlas Search condition
 */
function buildValueSearchCondition(condition, strategyConfig) {
  const { path } = condition.pathComponents;

  // Handle different patterns of value paths
  if (path.endsWith('/value/value')) {
    // Check if text search is enabled for this field
    const textSearchEnabled = isTextSearchEnabledForPath(path, strategyConfig);

    if (textSearchEnabled && condition.operator === '=') {
      // Use text search for text fields with equals operator
      return {
        embeddedDocument: {
          path: "sn",
          operator: {
            text: {
              query: condition.mongoValue,
              path: "sn.d.v",
              fuzzy: getFuzzyOptions(strategyConfig)
            }
          }
        }
      };
    } else {
      // Use equals for other cases
      return {
        embeddedDocument: {
          path: "sn",
          operator: {
            [getSearchOperator(condition.operator)]: {
              path: "sn.d.v",
              [getSearchOperatorArg(condition.operator)]: condition.mongoValue
            }
          }
        }
      };
    }
  }
  // Handle coded values (defining_code/code_string)
  else if (path.endsWith('/value/defining_code/code_string')) {
    // Special handling for MATCHES operator with code_string
    if (condition.operator === 'MATCHES') {
      // For MATCHES with multiple values, use should (logical OR)
      const shouldConditions = [];
      condition.mongoValue.forEach(value => {
        shouldConditions.push({
          equals: {
            path: "sn.d.cs",
            value: value
          }
        });
      });

      return {
        embeddedDocument: {
          path: "sn",
          operator: {
            compound: {
              should: shouldConditions,
              minimumShouldMatch: 1
            }
          }
        }
      };
    } else {
      // Normal operator for code_string
      return {
        embeddedDocument: {
          path: "sn",
          operator: {
            equals: {
              path: "sn.d.cs",
              value: condition.mongoValue
            }
          }
        }
      };
    }
  }

  // Default case if no specific handling
  return null;
}

/**
 * Build a time search condition for time-related filters.
 * 
 * @param {Object} condition - The WHERE condition
 * @returns {Object} Atlas Search condition
 */
function buildTimeSearchCondition(condition) {
  const { path } = condition.pathComponents;

  // Handle different time fields
  let searchPath = "";

  if (path === 'time/value') {
    searchPath = "sn.d.time";
  } else if (path === 'context/start_time/value') {
    searchPath = "sn.d.st";
  } else if (path.includes('date') || path.includes('time')) {
    // Generic time/date field
    searchPath = sn.d.${path.replace(/\//g, '.')};
  } else {
    // Can't handle this time path
    return null;
  }

  // Build the range condition
  return {
    embeddedDocument: {
      path: "sn",
      operator: {
        range: {
          path: searchPath,
          [getSearchRangeOperator(condition.operator)]: condition.mongoValue
        }
      }
    }
  };
}

/**
 * Build an archetype search condition for archetype-related filters.
 * 
 * @param {Object} condition - The WHERE condition
 * @returns {Object} Atlas Search condition
 */
function buildArchetypeSearchCondition(condition) {
  const { path } = condition.pathComponents;

  // Handle archetype_node_id condition
  if (path === 'archetype_node_id') {
    return {
      embeddedDocument: {
        path: "sn",
        operator: {
          equals: {
            path: "sn.d.ani",
            value: condition.mongoValue
          }
        }
      }
    };
  }
  // Handle archetype_details.archetype_id.value condition
  else if (path === 'archetype_details/archetype_id/value') {
    return {
      embeddedDocument: {
        path: "sn",
        operator: {
          equals: {
            path: "sn.d.ai",
            value: condition.mongoValue
          }
        }
      }
    };
  }

  // Can't handle other archetype paths
  return null;
}

/**
 * Check if a condition can be handled by Atlas Search.
 * 
 * @param {Object} condition - The WHERE condition
 * @param {Object} strategyConfig - Configuration options
 * @returns {boolean} True if the condition can be handled by Atlas Search
 */
function isSearchableCondition(condition, strategyConfig) {
  // Check if the operator is supported by Atlas Search
  const supportedOperators = ['=', '!=', '>', '>=', '<', '<=', 'MATCHES'];
  if (!supportedOperators.includes(condition.operator)) {
    return false;
  }

  // Check if the path is indexed for search
  const { path } = condition.pathComponents;

  // Common searchable paths
  const commonSearchablePaths = [
    '/archetype_node_id',
    '/value/value',
    '/value/defining_code/code_string',
    '/time/value',
    '/context/start_time/value'
  ];

  // First check if it's a common searchable path
  for (const searchablePath of commonSearchablePaths) {
    if (path.endsWith(searchablePath)) {
      return true;
    }
  }

  // Check config for additional searchable paths
  if (strategyConfig.atlasSearch && strategyConfig.atlasSearch.searchablePaths) {
    for (const searchablePath of strategyConfig.atlasSearch.searchablePaths) {
      if (path.includes(searchablePath)) {
        return true;
      }
    }
  }

  // Not searchable
  return false;
}

/**
 * Check if text search is enabled for a specific path.
 * 
 * @param {string} path - The path to check
 * @param {Object} strategyConfig - Configuration options
 * @returns {boolean} True if text search is enabled for this path
 */
function isTextSearchEnabledForPath(path, strategyConfig) {
  if (!strategyConfig.atlasSearch || !strategyConfig.atlasSearch.textSearchPaths) {
    return false;
  }

  // Check if this path is explicitly listed as a text search path
  for (const textPath of strategyConfig.atlasSearch.textSearchPaths) {
    if (path.endsWith(textPath)) {
      return true;
    }
  }

  // Not a text search path
  return false;
}

/**
 * Get fuzzy search options from strategy config.
 * 
 * @param {Object} strategyConfig - Configuration options
 * @returns {Object|undefined} Fuzzy search options or undefined if disabled
 */
function getFuzzyOptions(strategyConfig) {
  if (strategyConfig.atlasSearch &&
    strategyConfig.atlasSearch.fuzzySearch === false) {
    return undefined;
  }

  return {
    maxEdits: strategyConfig.atlasSearch?.fuzzyMaxEdits || 1,
    prefixLength: strategyConfig.atlasSearch?.fuzzyPrefixLength || 2
  };
}

/**
 * Map AQL comparison operators to Atlas Search range operators.
 * 
 * @param {string} operator - AQL operator
 * @returns {string} Atlas Search range operator
 */
function getSearchRangeOperator(operator) {
  const operatorMap = {
    '>': 'gt',
    '>=': 'gte',
    '<': 'lt',
    '<=': 'lte',
    '=': 'eq',
    '==': 'eq',
    '!=': 'ne'
  };

  return operatorMap[operator] || 'eq';
}

/**
 * Get the appropriate Atlas Search operator based on AQL operator.
 * 
 * @param {string} operator - AQL operator
 * @returns {string} Atlas Search operator
 */
function getSearchOperator(operator) {
  if (['>', '>=', '<', '<='].includes(operator)) {
    return 'range';
  } else if (operator === '!=' || operator === '<>') {
    return 'compound';
  } else {
    return 'equals';
  }
}

/**
 * Get the argument name for the search operator.
 * 
 * @param {string} operator - AQL operator
 * @returns {string} Argument name
 */
function getSearchOperatorArg(operator) {
  if (operator === '>') return 'gt';
  if (operator === '>=') return 'gte';
  if (operator === '<') return 'lt';
  if (operator === '<=') return 'lte';
  return 'value';
}

/**
 * Map openEHR RM type to its shorthand representation.
 * 
 * @param {string} rmType - RM type (e.g., "COMPOSITION", "EVALUATION")
 * @returns {string} Shorthand (e.g., "C", "E")
 */
function mapRmTypeToShorthand(rmType) {
  const rmTypeMap = {
    'COMPOSITION': 'C',
    'SECTION': 'S',
    'ADMIN_ENTRY': 'M',
    'OBSERVATION': 'O',
    'EVALUATION': 'E',
    'INSTRUCTION': 'I',
    'ACTION': 'A',
    'CLUSTER': 'K',
    'ITEM_TREE': 'T',
    'ITEM_LIST': 'L',
    'ITEM_SINGLE': 'G',
    'ITEM_TABLE': 'B',
    'ELEMENT': 'U',
    'HISTORY': 'H',
    'EVENT': 'V',
    'POINT_EVENT': 'P',
    'INTERVAL_EVENT': 'N',
    'ACTIVITY': 'Y',
    'ISM_TRANSITION': 'X',
    'INSTRUCTION_DETAILS': 'Z',
    'CARE_ENTRY': 'R',
    'PARTY_PROXY': 'W',
    'EVENT_CONTEXT': 'J'
  };

  return rmTypeMap[rmType] || rmType;
}

import {

  processValueForMongo, 
  extractArchetypeIdsFromPath 
} from '../../path-utils';

  mapOperatorToMongo 
} from '../../dictionary';

/**
 * Parse the WHERE clause of an AQL query to extract conditions.
 * 
 * @param {Object} ast - The AQL Abstract Syntax Tree
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @returns {Object} Information from the WHERE conditions
 */
export function parseWhereConditions(ast, containsInfo) {
  // Create default structure
  const whereInfo = {
    conditions: [],             // All conditions (flat list)
    ehrIdCondition: null,       // The ehr_id condition if present
    archetypeConditions: [],    // Conditions on archetype properties
    pathConditions: {},         // Conditions organized by path
    valueConditions: [],        // Conditions on values
    timeConditions: [],         // Conditions on timestamps
    logicalGroups: [],          // AND/OR logical groups
    unhandledConditions: []     // Conditions that couldn't be processed
  };
  
  // Return default if no WHERE clause
  if (!ast || !ast.where) {
    return whereInfo;
  }
  
  try {
    // Process the WHERE clause based on its structure
    if (ast.where.path && ast.where.operator) {
      // Single condition
      processSingleCondition(ast.where, whereInfo, containsInfo);
    } else if (ast.where.operator && ast.where.conditions) {
      // Logical group (AND/OR)
      processLogicalGroup(ast.where, whereInfo, containsInfo);
    }
    
    // Check for incomplete conditions (missing operator/value)
    fixIncompleteConditions(whereInfo);
    
  } catch (error) {
    console.error("[WHERE Parser] Error parsing WHERE conditions:", error);
  }
  
  return whereInfo;
}

/**
 * Process a single WHERE condition.
 * 
 * @param {Object} condition - The WHERE condition from the AST
 * @param {Object} whereInfo - Object being populated with WHERE information
 * @param {Object} containsInfo - Information from the CONTAINS clause
 */
function processSingleCondition(condition, whereInfo, containsInfo) {
  // Parse the path to extract alias and path components
  const { alias, path } = parseAqlPath(condition.path);
  
  // Create parsed condition with all necessary properties
  const parsedCondition = {
    path: condition.path,
    pathComponents: { alias, path },
    operator: condition.operator,
    mongoOperator: mapOperatorToMongo(condition.operator),
    value: condition.value,
    mongoValue: processValueForMongo(condition.value, condition.operator),
    archetypeIds: extractArchetypeIdsFromPath(path || ''),
    isEhrIdCondition: false,
    isArchetypeCondition: false,
    isTimeCondition: false,
    isValueCondition: false
  };
  
  // Determine the type of condition
  if (condition.path.includes('ehr_id/value')) {
    parsedCondition.isEhrIdCondition = true;
    whereInfo.ehrIdCondition = parsedCondition;
  } else if (condition.path.includes('archetype_node_id') || condition.path.includes('archetype_details')) {
    parsedCondition.isArchetypeCondition = true;
    whereInfo.archetypeConditions.push(parsedCondition);
  } else if (condition.path.includes('/time/value') || 
             condition.path.includes('start_time') || 
             condition.path.includes('date')) {
    parsedCondition.isTimeCondition = true;
    whereInfo.timeConditions.push(parsedCondition);
  } else if (condition.path.includes('/value/value') || 
             condition.path.includes('/value/magnitude') || 
             condition.path.includes('/value/defining_code/code_string')) {
    parsedCondition.isValueCondition = true;
    whereInfo.valueConditions.push(parsedCondition);
  } else {
    whereInfo.unhandledConditions.push(parsedCondition);
  }
  
  // Add to main conditions list
  whereInfo.conditions.push(parsedCondition);
  
  // Add to pathConditions for path-based lookup
  if (!whereInfo.pathConditions[condition.path]) {
    whereInfo.pathConditions[condition.path] = [];
  }
  whereInfo.pathConditions[condition.path].push(parsedCondition);
}

/**
 * Process a logical group (AND/OR) of conditions.
 * 
 * @param {Object} group - The logical group from the AST
 * @param {Object} whereInfo - Object being populated with WHERE information
 * @param {Object} containsInfo - Information from the CONTAINS clause
 */
function processLogicalGroup(group, whereInfo, containsInfo) {
  const logicalGroup = {
    operator: group.operator,
    mongoOperator: group.operator === 'AND' ? '$and' : '$or',
    conditions: []
  };
  
  // Process each condition in the group
  Object.values(group.conditions).forEach(condition => {
    if (condition.path && condition.operator) {
      // Single condition within the group
      processSingleCondition(condition, whereInfo, containsInfo);
      
      // Add to the logical group's conditions
      logicalGroup.conditions.push(condition);
    } else if (condition.operator && condition.conditions) {
      // Nested logical group
      const nestedGroup = {
        operator: condition.operator,
        mongoOperator: condition.operator === 'AND' ? '$and' : '$or',
        conditions: []
      };
      
      // Process the nested group recursively
      processLogicalGroup(condition, whereInfo, containsInfo);
      
      // Add to the parent logical group
      logicalGroup.conditions.push(nestedGroup);
    }
  });
  
  // Add to logical groups if not empty
  if (logicalGroup.conditions.length > 0) {
    whereInfo.logicalGroups.push(logicalGroup);
  }
}

/**
 * Fix incomplete conditions by filling in missing information.
 * This is helpful for handling potentially incomplete AST data.
 * 
 * @param {Object} whereInfo - Object with WHERE information to be fixed
 */
function fixIncompleteConditions(whereInfo) {
  // Find conditions missing operator or value
  whereInfo.conditions.forEach((condition, index) => {
    // Check if operator is missing
    if (!condition.operator || condition.operator === null) {
      console.warn([WHERE Parser] Condition at index ${index} is missing operator, applying default '=');
      condition.operator = '=';
      condition.mongoOperator = '$eq';
    }
    
    // Check if value is missing or null
    if (condition.value === null || condition.value === undefined) {
      console.warn([WHERE Parser] Condition at index ${index} is missing value, applying default null);
      condition.value = null;
      condition.mongoValue = null;
    }
    
    // Handle MATCHES with code_string - a common pattern in openEHR queries
    if (condition.operator === 'MATCHES' && 
        condition.path && 
        condition.path.endsWith('/value/defining_code/code_string')) {
      
      // Ensure MATCHES values are properly formatted as arrays
      if (!Array.isArray(condition.mongoValue)) {
        if (typeof condition.mongoValue === 'string') {
          // Handle the {'val1','val2'} format
          if (condition.mongoValue.startsWith('{') && condition.mongoValue.endsWith('}')) {
            const innerValue = condition.mongoValue.substring(1, condition.mongoValue.length - 1);
            condition.mongoValue = innerValue.split(',')
              .map(v => v.trim())
              .map(v => v.startsWith("'") && v.endsWith("'") ? v.substring(1, v.length - 1) : v);
          } else {
            condition.mongoValue = [condition.mongoValue];
          }
        } else {
          condition.mongoValue = [condition.mongoValue];
        }
      }
    }
  });
}

export default parseWhereConditions;

import { buildVariableNameForPath as buildVarName } from '../../path-utils';

/**
 * Build the MongoDB $addFields stage to extract referenced entities and elements.
 * 
 * @param {Object} ast - The AQL Abstract Syntax Tree
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @param {string} nodeField - Field name for the nodes array ("cn" or "sn")
 * @returns {Object} MongoDB $addFields stage
 */
export function buildAddFieldsStage(ast, containsInfo, nodeField = "cn") {
  // Initialize the addFields stage
  const addFields = {};
  
  // First, extract all the entities referenced in CONTAINS clauses
  buildEntityExtractions(containsInfo, nodeField, addFields);
  
  // Then extract specific paths referenced in SELECT, WHERE, ORDER BY clauses
  if (ast.select) {
    buildSelectPathExtractions(ast.select, containsInfo, nodeField, addFields);
  }
  
  if (ast.where) {
    buildWherePathExtractions(ast.where, containsInfo, nodeField, addFields);
  }
  
  if (ast.orderBy) {
    buildOrderByPathExtractions(ast.orderBy, containsInfo, nodeField, addFields);
  }
  
  return addFields;
}

/**
 * Build extractions for entities referenced in CONTAINS clauses.
 * 
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @param {string} nodeField - Field name for the nodes array
 * @param {Object} addFields - Object being populated with extraction expressions
 */
function buildEntityExtractions(containsInfo, nodeField, addFields) {
  // Process each entry in CONTAINS
  if (containsInfo.entries && containsInfo.entries.length > 0) {
    containsInfo.entries.forEach(entry => {
      if (entry.alias && (entry.rmType || entry.archetypeId)) {
        // Build an extraction for this entity
        const conditions = [];
        
        // Add type condition if available
        if (entry.rmType) {
          conditions.push({ $eq: ["$$node.d.T", mapRmTypeToShorthand(entry.rmType)] });
        }
        
        // Add archetype ID condition if available
        if (entry.archetypeId) {
          conditions.push({ 
            $or: [
              { $eq: ["$$node.d.ani", entry.archetypeId] },
              { $eq: ["$$node.d.ad.ai.v", entry.archetypeId] }
            ]
          });
        }
        
        // Add parent relationship condition if this is a nested entity
        if (entry.parentAlias && containsInfo.entriesByAlias && containsInfo.entriesByAlias[entry.parentAlias]) {
          const parentEntry = containsInfo.entriesByAlias[entry.parentAlias];
          if (parentEntry.archetypeId) {
            conditions.push({ 
              $anyElementTrue: {
                $map: {
                  input: { $ifNull: ["$$node.a", []] },
                  as: "ancestor",
                  in: { $eq: ["$$ancestor", parentEntry.archetypeId] }
                }
              }
            });
          }
        }
        
        // Build the $filter expression to extract this entity
        addFields[entry.alias] = {
          $arrayElemAt: [
            {
              $filter: {
                input: $${nodeField},
                as: "node",
                cond: { $and: conditions }
              }
            },
            0
          ]
        };
      }
    });
  }
}

/**
 * Build extractions for paths referenced in SELECT clause.
 * 
 * @param {Object} select - SELECT clause from the AST
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @param {string} nodeField - Field name for the nodes array
 * @param {Object} addFields - Object being populated with extraction expressions
 */
function buildSelectPathExtractions(select, containsInfo, nodeField, addFields) {
  // Extract columns from the SELECT clause
  const columns = Array.isArray(select.columns) 
    ? select.columns 
    : (typeof select.columns === 'object' ? Object.values(select.columns) : []);
  
  // Process each column
  columns.forEach(column => {
    let path = '';
    
    // Extract the path based on the column format
    if (column.value && column.value.path) {
      path = column.value.path;
    } else if (column.path) {
      path = column.path;
    } else if (column.aqlPath) {
      path = column.aqlPath;
    }
    
    if (path) {
      // Extract alias and path components
      const pathParts = path.split('/');
      if (pathParts.length >= 2) {
        const alias = pathParts[0];
        const restPath = pathParts.slice(1).join('/');
        
        // Check if this is a complex path with archetype IDs
        if (restPath.includes('[') && restPath.includes(']')) {
          buildComplexPathExtraction(alias, restPath, nodeField, addFields, containsInfo);
        }
      }
    }
  });
}

/**
 * Build extractions for paths referenced in WHERE clause.
 * 
 * @param {Object} where - WHERE clause from the AST
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @param {string} nodeField - Field name for the nodes array
 * @param {Object} addFields - Object being populated with extraction expressions
 */
function buildWherePathExtractions(where, containsInfo, nodeField, addFields) {
  // Skip ehr_id conditions as they're not part of the nodes
  if (where.path && !where.path.includes('ehr_id')) {
    const pathParts = where.path.split('/');
    if (pathParts.length >= 2) {
      const alias = pathParts[0];
      const restPath = pathParts.slice(1).join('/');
      
      // Check if this is a complex path with archetype IDs
      if (restPath.includes('[') && restPath.includes(']')) {
        buildComplexPathExtraction(alias, restPath, nodeField, addFields, containsInfo);
      }
    }
  }
  
  // Process logical groups (AND/OR)
  if (where.conditions) {
    Object.values(where.conditions).forEach(condition => {
      if (condition.path && !condition.path.includes('ehr_id')) {
        const pathParts = condition.path.split('/');
        if (pathParts.length >= 2) {
          const alias = pathParts[0];
          const restPath = pathParts.slice(1).join('/');
          
          // Check if this is a complex path with archetype IDs
          if (restPath.includes('[') && restPath.includes(']')) {
            buildComplexPathExtraction(alias, restPath, nodeField, addFields, containsInfo);
          }
        }
      } else if (condition.operator && condition.conditions) {
        // Recursively process nested logical groups
        buildWherePathExtractions(condition, containsInfo, nodeField, addFields);
      }
    });
  }
}

/**
 * Build extractions for paths referenced in ORDER BY clause.
 * 
 * @param {Object} orderBy - ORDER BY clause from the AST
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @param {string} nodeField - Field name for the nodes array
 * @param {Object} addFields - Object being populated with extraction expressions
 */
function buildOrderByPathExtractions(orderBy, containsInfo, nodeField, addFields) {
  // Extract sort items from the ORDER BY clause
  const sortItems = Array.isArray(orderBy) 
    ? orderBy 
    : (typeof orderBy === 'object' ? Object.values(orderBy) : []);
  
  // Process each sort item
  sortItems.forEach(item => {
    let path = '';
    
    // Extract the path based on the item format
    if (item.path) {
      path = item.path;
    } else if (item.aqlPath) {
      path = item.aqlPath;
    }
    
    if (path) {
      // Extract alias and path components
      const pathParts = path.split('/');
      if (pathParts.length >= 2) {
        const alias = pathParts[0];
        const restPath = pathParts.slice(1).join('/');
        
        // Check if this is a complex path with archetype IDs
        if (restPath.includes('[') && restPath.includes(']')) {
          buildComplexPathExtraction(alias, restPath, nodeField, addFields, containsInfo);
        }
      }
    }
  });
}

/**
 * Build extraction for a complex path with archetype IDs.
 * This handles paths like "ev/data[at0001]/items[at0002]/value/value"
 * by creating nested filters to extract each archetype node.
 * 
 * @param {string} alias - Path alias (e.g., "ev")
 * @param {string} path - Path after the alias (e.g., "data[at0001]/items[at0002]/value/value")
 * @param {string} nodeField - Field name for the nodes array
 * @param {Object} addFields - Object being populated with extraction expressions
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 */
function buildComplexPathExtraction(alias, path, nodeField, addFields, containsInfo) {
  // Generate a variable name for this path
  const varName = buildVarName(alias, path);
  
  // Don't duplicate extractions
  if (addFields[varName]) {
    return;
  }
  
  // Extract archetype IDs from the path
  const archetypeMatches = path.match(/\[(at\d+|openEHR-EHR-[A-Z_]+\.[a-z_]+\.v\d+)\]/g) || [];
  const archetypeIds = archetypeMatches.map(match => match.substring(1, match.length - 1));
  
  // Get the root entity's archetype ID
  let parentArchetypeId = null;
  if (containsInfo.entriesByAlias && containsInfo.entriesByAlias[alias]) {
    parentArchetypeId = containsInfo.entriesByAlias[alias].archetypeId;
  }
  
  // Build the extraction expression
  if (archetypeIds.length > 0) {
    const conditions = [];
    
    // Add condition for the last archetype ID
    const lastArchetypeId = archetypeIds[archetypeIds.length - 1];
    conditions.push({ 
      $or: [
        { $eq: ["$$node.d.ani", lastArchetypeId] },
        { $eq: ["$$node.d.ad.ai.v", lastArchetypeId] }
      ]
    });
    
    // Add condition for the parent entity
    if (parentArchetypeId) {
      conditions.push({ 
        $anyElementTrue: {
          $map: {
            input: { $ifNull: ["$$node.a", []] },
            as: "ancestor",
            in: { $eq: ["$$ancestor", parentArchetypeId] }
          }
        }
      });
    }
    
    // Add conditions for intermediate archetype IDs if present
    if (archetypeIds.length > 1) {
      for (let i = 0; i < archetypeIds.length - 1; i++) {
        const archetypeId = archetypeIds[i];
        conditions.push({ 
          $anyElementTrue: {
            $map: {
              input: { $ifNull: ["$$node.a", []] },
              as: "ancestor",
              in: { $eq: ["$$ancestor", archetypeId] }
            }
          }
        });
      }
    }
    
    // Build the $filter expression
    addFields[varName] = {
      $arrayElemAt: [
        {
          $filter: {
            input: $${nodeField},
            as: "node",
            cond: { $and: conditions }
          }
        },
        0
      ]
    };
  }
}

/**
 * Build a variable name for a path that can be used in MongoDB aggregation pipeline.
 * This is a wrapper around the utility function to ensure consistency.
 * 
 * @param {string} alias - Path alias
 * @param {string} path - AQL path after the alias
 * @returns {string} Variable name safe for MongoDB
 */
export function buildVariableNameForPath(alias, path) {
  return buildVarName(alias, path);
}

/**
 * Build the MongoDB $project stage for the SELECT clause.
 * 
 * @param {Object} ast - The AQL Abstract Syntax Tree
 * @param {Object} containsInfo - Information from the parsed CONTAINS clauses
 * @returns {Object} MongoDB $project stage
 */
export function buildProjectStage(ast, containsInfo) {
  // Initialize the project stage
  const projectStage = {};
  
  // Always include _id in the result
  projectStage._id = 1;
  
  // Return empty if there's no SELECT clause
  if (!ast || !ast.select) {
    return projectStage;
  }
  
  // Extract columns from the SELECT clause
  const columns = Array.isArray(ast.select.columns) 
    ? ast.select.columns 
    : (typeof ast.select.columns === 'object' ? Object.values(ast.select.columns) : []);
  
  // Process each column
  columns.forEach(column => {
    let path = '';
    let alias = '';
    
    // Extract the path and alias based on the column format
    if (column.value && column.value.path) {
      path = column.value.path;
      alias = column.alias || column.name || '';
    } else if (column.path) {
      path = column.path;
      alias = column.alias || column.name || '';
    } else if (column.aqlPath) {
      path = column.aqlPath;
      alias = column.aqlColumn || '';
    }
    
    if (path) {
      // Build the projection for this column
      buildColumnProjection(path, alias, projectStage);
    }
  });
  
  return projectStage;
}

/**
 * Build a projection for a SELECT column.
 * 
 * @param {string} path - AQL path (e.g., "ev/data[at0001]/items[at0002]/value/value")
 * @param {string} alias - Column alias or name
 * @param {Object} projectStage - Object being populated with projections
 */
function buildColumnProjection(path, alias, projectStage) {
  // Split the path to get the alias and the rest
  const pathParts = path.split('/');
  
  if (pathParts.length < 2) {
    // Simple field, just include it
    projectStage[alias || path] = ${path};
    return;
  }
  
  const entityAlias = pathParts[0];
  const restPath = pathParts.slice(1).join('/');
  
  // Use the alias if provided, otherwise generate a field name from the path
  const outputField = alias || path.replace(/\//g, '_').replace(/[\[\]]/g, '_');
  
  // Check if this is a complex path with archetype IDs
  if (restPath.includes('[') && restPath.includes(']')) {
    buildComplexPathProjection(entityAlias, restPath, outputField, projectStage);
  } else {
    buildSimplePathProjection(entityAlias, restPath, outputField, projectStage);
  }
}

/**
 * Build a projection for a complex path with archetype IDs.
 * 
 * @param {string} entityAlias - Entity alias (e.g., "ev")
 * @param {string} path - Path after the entity alias (e.g., "data[at0001]/items[at0002]/value/value")
 * @param {string} outputField - Field name for the output
 * @param {Object} projectStage - Object being populated with projections
 */
function buildComplexPathProjection(entityAlias, path, outputField, projectStage) {
  // Generate a variable name for this path (used in $addFields)
  const varName = buildVariableNameForPath(entityAlias, path);
  
  // Check if path ends with /value or /value/value
  if (path.endsWith('/value/value')) {
    projectStage[outputField] = ${varName}.d.v.v;
  } else if (path.endsWith('/value')) {
    projectStage[outputField] = ${varName}.d.v;
  } else if (path.endsWith('/value/defining_code/code_string')) {
    projectStage[outputField] = ${varName}.d.v.df.cs;
  } else if (path.endsWith('/value/magnitude')) {
    projectStage[outputField] = ${varName}.d.v.magnitude;
  } else {
    // Extract the last segment for a generic case
    const segments = path.split('/');
    const lastSegment = segments[segments.length - 1];
    
    // Remove any archetype ID
    const cleanSegment = lastSegment.replace(/\[[^\]]*\]/g, '');
    
    // We'll try a few common patterns
    projectStage[outputField] = {
      $ifNull: [
        ${varName}.d.${mapFieldToShorthand(cleanSegment)},
        ${varName}.d.v.${mapFieldToShorthand(cleanSegment)},
        ${varName}.d.${mapFieldToShorthand(cleanSegment)}.v,
        null
      ]
    };
  }
}

/**
 * Build a projection for a simple path without archetype IDs.
 * 
 * @param {string} entityAlias - Entity alias (e.g., "ev")
 * @param {string} path - Path after the entity alias (e.g., "uid/value")
 * @param {string} outputField - Field name for the output
 * @param {Object} projectStage - Object being populated with projections
 */
function buildSimplePathProjection(entityAlias, path, outputField, projectStage) {
  // Handle some common paths
  if (path === 'uid/value') {
    projectStage[outputField] = ${entityAlias}.d.uid.v;
  } else if (path === 'time/value') {
    projectStage[outputField] = ${entityAlias}.d.time.v;
  } else if (path === 'context/start_time/value') {
    projectStage[outputField] = ${entityAlias}.d.cx.st.v;
  } else {
    // Convert the path to dot notation with shortened field names
    const segments = path.split('/');
    const mongoPath = segments.map(mapFieldToShorthand).join('.');
    
    projectStage[outputField] = ${entityAlias}.d.${mongoPath};
  }
}

/**
 * Map field names to their shorthand forms.
 * 
 * @param {string} field - Field name
 * @returns {string} Shorthand form of the field name
 */
function mapFieldToShorthand(field) {
  const fieldMap = {
    'value': 'v',
    'magnitude': 'magnitude',
    'units': 'units',
    'name': 'n',
    'archetype_node_id': 'ani',
    'archetype_details': 'ad',
    'context': 'cx',
    'items': 'i',
    'defining_code': 'df',
    'code_string': 'cs',
    'terminology_id': 'tid'
  };
  
  return fieldMap[field] || field;
}

/**
 * Handle aggregate functions in the SELECT clause.
 * 
 * @param {Object} aggregateFunc - Aggregate function information
 * @param {string} outputField - Field name for the output
 * @param {Object} projectStage - Object being populated with projections
 */
function buildAggregateProjection(aggregateFunc, outputField, projectStage) {
  if (!aggregateFunc || !aggregateFunc.name) {
    return;
  }
  
  const funcName = aggregateFunc.name.toUpperCase();
  const path = aggregateFunc.args || '';
  
  // Handle different types of aggregate functions
  switch (funcName) {
    case 'COUNT':
      // For COUNT, we use $size on an array
      projectStage[outputField] = { $size: getPathForCount(path) };
      break;
    
    case 'MAX':
      projectStage[outputField] = { $max: getPathForAggregate(path) };
      break;
    
    case 'MIN':
      projectStage[outputField] = { $min: getPathForAggregate(path) };
      break;
    
    case 'AVG':
      projectStage[outputField] = { $avg: getPathForAggregate(path) };
      break;
    
    case 'SUM':
      projectStage[outputField] = { $sum: getPathForAggregate(path) };
      break;
    
    default:
      // Unknown aggregate function
      projectStage[outputField] = null;
  }
}

/**
 * Get the MongoDB path for COUNT function.
 * 
 * @param {string} path - The path to count
 * @returns {string} MongoDB expression for COUNT
 */
function getPathForCount(path) {
  // Remove surrounding brackets if present in the path
  const cleanPath = path.replace(/^\(|\)$/g, '').trim();
  
  // Split the path to get the alias and the rest
  const pathParts = cleanPath.split('/');
  
  if (pathParts.length < 2) {
    // Simple field
    return ${cleanPath};
  }
  
  const entityAlias = pathParts[0];
  const restPath = pathParts.slice(1).join('/');
  
  // For complex paths with archetype IDs, we might need a different approach
  // Here we just return a simple array reference as an example
  return ${entityAlias}_${restPath.replace(/\//g, '_').replace(/[\[\]]/g, '_')};
}

/**
 * Get the MongoDB path for other aggregate functions.
 * 
 * @param {string} path - The path for aggregation
 * @returns {string} MongoDB expression for aggregation
 */
function getPathForAggregate(path) {
  // Similar logic to getPathForCount, but might need adjustments
  // based on the specific aggregate function and path format
  return getPathForCount(path);
}

/**
 * Build MongoDB stages for query options (ORDER BY, LIMIT, OFFSET).
 * 
 * @param {Object} ast - The AQL Abstract Syntax Tree
 * @returns {Array} Array of MongoDB pipeline stages
 */
export function buildOptionsStages(ast) {
  const stages = [];
  
  // Add sorting stage if ORDER BY is present
  if (ast.orderBy) {
    const sortStage = buildSortStage(ast.orderBy);
    if (sortStage && Object.keys(sortStage.$sort || {}).length > 0) {
      stages.push(sortStage);
    }
  }
  
  // Add LIMIT stage if present
  if (ast.limit !== null && ast.limit !== undefined) {
    const limit = parseInt(ast.limit, 10);
    if (!isNaN(limit) && limit > 0) {
      stages.push({ $limit: limit });
    }
  }
  
  // Add OFFSET (skip) stage if present
  if (ast.offset !== null && ast.offset !== undefined) {
    const offset = parseInt(ast.offset, 10);
    if (!isNaN(offset) && offset > 0) {
      stages.push({ $skip: offset });
    }
  }
  
  return stages;
}

/**
 * Build the MongoDB $sort stage for ORDER BY clause.
 * 
 * @param {Object} orderBy - ORDER BY clause from the AST
 * @returns {Object} MongoDB $sort stage
 */
function buildSortStage(orderBy) {
  const sort = {};
  
  // Extract sort items from the ORDER BY clause
  const sortItems = Array.isArray(orderBy) 
    ? orderBy 
    : (typeof orderBy === 'object' ? Object.values(orderBy) : []);
  
  // Process each sort item
  sortItems.forEach(item => {
    let path = '';
    let direction = 1; // Default: ascending
    
    // Extract path and direction based on the item format
    if (item.path) {
      path = item.path;
      direction = (item.direction && item.direction.toUpperCase() === 'DESC') ? -1 : 1;
    } else if (item.aqlPath) {
      path = item.aqlPath;
      direction = (item.direction && item.direction.toUpperCase() === 'DESC') ? -1 : 1;
    }
    
    if (path) {
      // Build the sort field
      const sortField = buildSortField(path);
      if (sortField) {
        sort[sortField] = direction;
      }
    }
  });
  
  return Object.keys(sort).length > 0 ? { $sort: sort } : {};
}

/**
 * Build a MongoDB sort field from an AQL path.
 * 
 * @param {string} path - AQL path (e.g., "ev/data[at0001]/items[at0002]/value/value")
 * @returns {string} MongoDB field for sorting
 */
function buildSortField(path) {
  // Split the path to get the alias and the rest
  const pathParts = path.split('/');
  
  if (pathParts.length < 2) {
    // Simple field
    return path;
  }
  
  const entityAlias = pathParts[0];
  const restPath = pathParts.slice(1).join('/');
  
  // Check if this is a complex path with archetype IDs
  if (restPath.includes('[') && restPath.includes(']')) {
    return buildComplexSortField(entityAlias, restPath);
  } else {
    return buildSimpleSortField(entityAlias, restPath);
  }
}
